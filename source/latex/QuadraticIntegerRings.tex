\section{Quadratic integer rings}

\begin{snippettheorem}{quadratic-field-ring-of-integers}{Ring of integers of a quadratic field}
    Let \(d\in\integers\) be square-free, with \(d\neq0,1\). The ring of
    algebraic integers in the quadratic \field
    \(\rationalnumbers(\sqrt d)\) is
    \[
        \mathcal{O}_{\rationalnumbers(\sqrt d)}
        =
        \begin{cases}
            \integers[\sqrt d],
            & d\equiv2,3\pmod4,\\[4pt]
            \displaystyle
            \integers\left[\frac{1+\sqrt d}{2}\right],
            & d\equiv1\pmod4.
        \end{cases}
    \]
\end{snippettheorem}

\begin{snippetexample}{sqrt-minus-five-ring-of-integers-example}{The quadratic field with square root of minus five}
    Since \(-5\equiv3\pmod4\), the ring of algebraic integers of
    \(\rationalnumbers(\sqrt{-5})\) is
    \[
        \mathcal{O}_{\rationalnumbers(\sqrt{-5})}
        =
        \integers[\sqrt{-5}].
    \]
\end{snippetexample}
