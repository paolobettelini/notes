\section{The Lebesgue Integral}

\includesnpt{simple-function-definition}

If \(s\) is a nonnegative measurable simple function, then it can be written
using its finitely many values as
\[
    s=\sum_{i=1}^n\alpha_i\chi_{s^{-1}(\alpha_i)},
    \qquad \alpha_i\geq0.
\]

\begin{snippetdefinition}{nonnegative-simple-function-integral-definition}{Integral of a nonnegative simple function}
    Let \((X,\mathcal M,\mu)\) be a measure space, let \(s\) be a nonnegative
    measurable simple function, and let \(\Omega\in\mathcal M\). Define
    \[
        \int_\Omega s\,d\mu
        \triangleq
        \sum_{i=1}^n
        \alpha_i\mu\bigl(s^{-1}(\alpha_i)\cap\Omega\bigr).
    \]
\end{snippetdefinition}

Whereas a Riemann sum groups points into intervals in the domain, the
Lebesgue integral groups them into fibres determined by values in the
codomain. We use the convention \(0\cdot+\infty=0\). On the real line with
Lebesgue measure, Riemann step functions are measurable simple functions and
the two definitions of their integral agree.

\begin{snippetdefinition}{nonnegative-measurable-function-integral-definition}{Integral of a nonnegative measurable function}
    Let \(f\colon X\fromto[0,+\infty]\) be measurable and let
    \(\Omega\in\mathcal M\). Define
    \[
        \int_\Omega f\,d\mu
        \triangleq
        \sup_{\substack{s\text{ nonnegative}\\\text{measurable simple}\\0\leq s\leq f}}
        \int_\Omega s\,d\mu.
    \]
\end{snippetdefinition}

Equivalently, one may define integration only over the whole space and use
the restricted measure space on \(\Omega\), or the identity
\(\int_\Omega f\,d\mu=\int_Xf\chi_\Omega\,d\mu\). When \(f\) is simple, this
definition agrees with the preceding one because \(f\) itself is the largest
simple function below \(f\).

\begin{snippetproposition}{nonnegative-lebesgue-integral-basic-properties}{Basic properties of the nonnegative integral}
    Let \(f,g\colon X\fromto[0,+\infty]\) be measurable, let
    \(E,F\in\mathcal M\), and let \(\alpha\geq0\). Then:
    \begin{enumerate}
        \item if \(f\leq g\), then
            \[
                \int_Ef\,d\mu\leq\int_Eg\,d\mu;
            \]
        \item if \(F\subseteq E\), then
            \[
                \int_Ff\,d\mu\leq\int_Ef\,d\mu;
            \]
        \item
            \[
                \int_E\alpha f\,d\mu=\alpha\int_Ef\,d\mu;
            \]
        \item if \(f=0\) on \(E\), then \(\int_Ef\,d\mu=0\), even when
            \(\mu(E)=+\infty\);
        \item if \(\mu(E)=0\), then \(\int_Ef\,d\mu=0\), even when
            \(f=+\infty\) on \(E\);
        \item
            \[
                \int_Ef\,d\mu=\int_Xf\chi_E\,d\mu.
            \]
    \end{enumerate}
\end{snippetproposition}

Additivity is immediate neither from the supremum definition nor from
monotonicity. We first establish it for simple functions.

\begin{snippetproposition}{simple-function-induced-measure}{The measure induced by a simple function}
    Let \(s\) be a nonnegative measurable simple function. Then
    \[
        \varphi(E)\triangleq\int_Es\,d\mu,
        \qquad E\in\mathcal M,
    \]
    defines a measure on \(\mathcal M\).
\end{snippetproposition}

\begin{snippetproof}{simple-function-induced-measure-proof}{simple-function-induced-measure}{The measure induced by a simple function}
    Write \(s=\sum_{i=1}^n\alpha_i\chi_{A_i}\), where the \(A_i\) are
    measurable and pairwise disjoint. If \(\{E_k\}_{k\geq1}\) are pairwise
    disjoint measurable sets, then
    \begin{align*}
        \varphi\!\left(\bigcup_{k=1}^{\infty}E_k\right)
        &=\sum_{i=1}^n\alpha_i
          \mu\!\left(A_i\cap\bigcup_{k=1}^{\infty}E_k\right)\\
        &=\sum_{i=1}^n\alpha_i\sum_{k=1}^{\infty}\mu(A_i\cap E_k)\\
        &=\sum_{k=1}^{\infty}\varphi(E_k).
    \end{align*}
    Also \(\varphi(\emptyset)=0\), so \(\varphi\) is a measure.
\end{snippetproof}

\begin{snippetproposition}{nonnegative-simple-integral-additivity}{Additivity for nonnegative simple functions}
    If \(s,t\) are nonnegative measurable simple functions, then for every
    \(E\in\mathcal M\),
    \[
        \int_E(s+t)\,d\mu
        =\int_Es\,d\mu+\int_Et\,d\mu.
    \]
\end{snippetproposition}

\begin{snippetproof}{nonnegative-simple-integral-additivity-proof}{nonnegative-simple-integral-additivity}{Additivity for nonnegative simple functions}
    Write
    \[
        s=\sum_{i=1}^n\alpha_i\chi_{A_i},
        \qquad
        t=\sum_{j=1}^m\beta_j\chi_{B_j},
    \]
    with each family a measurable partition of \(X\), and put
    \(E_{ij}=E\cap A_i\cap B_j\). These sets form a finite measurable
    partition of \(E\), and on \(E_{ij}\) the functions \(s,t,s+t\) take the
    values \(\alpha_i,\beta_j,\alpha_i+\beta_j\), respectively. Hence
    \[
        \int_{E_{ij}}(s+t)\,d\mu
        =\int_{E_{ij}}s\,d\mu+\int_{E_{ij}}t\,d\mu.
    \]
    Summing over \(i,j\), using finite additivity of the induced measures,
    gives the result.
\end{snippetproof}

Passing from simple functions to arbitrary nonnegative measurable functions
requires a convergence theorem.
