\section{Polynomial rings}

\begin{snippetdefinition}{polynomial-definition}{Polynomial ring}
    Let \(A\) be a commutative \ring with identity. The \emph{polynomial
    ring} \(A[x]\) is the \set of sequences
    \[
        (a_0,a_1,a_2,\ldots)\in A^{\naturalnumbers}
    \]
    with finite support. Such a sequence is written formally as
    \[
        f=\sum_{i\geq0}a_ix^i,
    \]
    where \(x\) is an indeterminate. Addition is coefficientwise, while
    multiplication is defined by convolution:
    \[
        \left(\sum_{i\geq0}a_ix^i\right)
        \left(\sum_{j\geq0}b_jx^j\right)
        =
        \sum_{n\geq0}\left(\sum_{i+j=n}a_ib_j\right)x^n.
    \]
    The elements of \(A\) are identified with the constant polynomials,
    and \(x=(0_A,1_A,0_A,\ldots)\).
\end{snippetdefinition}

\begin{snippet}{polynomial-formal-expression-versus-function}
    Let \(A\) be a commutative \ring. A \polynomial in \(A[x]\) is a
    formal sequence of coefficients, not a
    \function. Every \(f\in A[x]\) induces the polynomial \function
    \[
        \widetilde f\colon A\fromto A,
        \qquad
        \widetilde f(a)=\sum_{i\geq0}a_i a^i,
    \]
    but distinct polynomials may induce the same function. For example,
    \(x^2-x\) and the zero polynomial induce the same function on
    \(\integers/2\integers\).
\end{snippet}

\begin{snippetdefinition}{polynomial-degree-definition}{Polynomial degree}
    Let \(A\) be a commutative \ring and let
    \[
        f=\sum_{i\geq0}a_ix^i\in A[x]
    \]
    be nonzero. Its \emph{degree} and \emph{leading coefficient} are
    \[
        \polynomialdeg f=\max\{i\in\naturalnumbers\suchthat a_i\neq0_A\},
        \qquad
        \operatorname{lc}(f)=a_{\polynomialdeg f}.
    \]
    A nonzero polynomial is \emph{monic} if
    \(\operatorname{lc}(f)=1_A\).
\end{snippetdefinition}

\begin{snippet}{null-polynomial-expl}
    Let \(A\) be a commutative \ring. The zero polynomial in \(A[x]\)
    has no degree. It is often convenient to adopt
    the convention
    \[
        \polynomialdeg 0=-\infty,
    \]
    which makes standard degree inequalities valid without a separate
    zero case.
\end{snippet}

\begin{snippetproposition}{polynomial-degree-arithmetic}{Degree and arithmetic}
    Let \(A\) be a commutative \ring and let \(f,g\in A[x]\). Then
    \[
        \polynomialdeg(f+g)
        \leq
        \max\{\polynomialdeg f,\polynomialdeg g\}.
    \]
    If \(f,g\neq0\), then
    \[
        \polynomialdeg(fg)\leq\polynomialdeg f+\polynomialdeg g.
    \]
    Equality holds whenever
    \(\operatorname{lc}(f)\operatorname{lc}(g)\neq0_A\); in particular,
    it always holds when \(A\) is an \integraldomain.
\end{snippetproposition}

\begin{snippetdefinition}{multivariable-polynomial-ring-definition}{Polynomials in several indeterminates}
    Let \(A\) be a commutative \ring and let \(n\geq1\). The polynomial
    \ring in \(n\) commuting indeterminates is defined recursively by
    \[
        A[x_1,\ldots,x_n]
        =
        A[x_1,\ldots,x_{n-1}][x_n].
    \]
    Equivalently, its elements are finite formal sums
    \[
        \sum_{\alpha\in\naturalnumbers^n}a_\alpha x^\alpha,
        \qquad
        x^\alpha=x_1^{\alpha_1}\cdots x_n^{\alpha_n}.
    \]
\end{snippetdefinition}

\section{Polynomial division}

\begin{snippettheorem}{polynomial-division-theorem}{Polynomial division}
    Let \(A\) be a commutative \ring, and let \(f,g\in A[x]\), with
    \(g\neq0\) and \(\operatorname{lc}(g)\in A^\times\). There exist
    unique \polynomial[polynomials] \(q,r\in A[x]\) such that
    \[
        f=gq+r,
        \qquad
        r=0\text{ or }\polynomialdeg r<\polynomialdeg g.
    \]
    In particular, this holds for every nonzero \(g\) when \(A\) is a
    \field.
\end{snippettheorem}

\begin{snippetproof}{polynomial-division-proof}{polynomial-division-theorem}{Polynomial division}
    Set \(q_0=0\) and \(r_0=f\). If \(r_0=0\) or
    \(\polynomialdeg r_0<\polynomialdeg g\), the required decomposition
    is already obtained. Otherwise, write
    \[
        n=\polynomialdeg r_0,
        \qquad
        m=\polynomialdeg g,
        \qquad
        a=\operatorname{lc}(r_0),
        \qquad
        b=\operatorname{lc}(g).
    \]
    Since \(b\) is a unit, define
    \[
        q_1=q_0+ab^{-1}x^{n-m},
        \qquad
        r_1=f-gq_1.
    \]
    The leading terms cancel, so either \(r_1=0\) or
    \(\polynomialdeg r_1<\polynomialdeg r_0\). Repeating this step gives
    a strictly decreasing sequence of nonnegative degrees and therefore
    terminates, producing \(f=gq+r\) with the required bound.

    For uniqueness, suppose
    \[
        f=gq+r=gq'+r',
    \]
    where both remainders are zero or have degree smaller than
    \(\polynomialdeg g\). Then
    \[
        g(q-q')=r'-r.
    \]
    If \(q-q'\neq0\), the leading coefficient of \(g\) being a unit
    gives
    \[
        \polynomialdeg\bigl(g(q-q')\bigr)
        =
        \polynomialdeg g+\polynomialdeg(q-q')
        \geq
        \polynomialdeg g,
    \]
    whereas the right-hand side has degree smaller than
    \(\polynomialdeg g\), a contradiction. Hence \(q=q'\), and then
    \(r=r'\).
\end{snippetproof}

\section{Evaluation and roots}

\begin{snippetproposition}{polynomial-evaluation-homomorphism}{Evaluation homomorphism}
    Let \(A\) be a commutative \ring and let \(a\in A\). Evaluation at
    \(a\) is the \ringhomomorphism
    \[
        \operatorname{ev}_a\colon A[x]\fromto A,
        \qquad
        \operatorname{ev}_a(f)=f(a),
    \]
    which restricts to the identity on \(A\).
\end{snippetproposition}

\begin{snippettheorem}{polynomial-remainder-theorem}{Remainder theorem}
    Let \(A\) be a commutative \ring, \(f\in A[x]\), and \(a\in A\).
    There exists a unique \(q\in A[x]\) such that
    \[
        f(x)=(x-a)q(x)+f(a).
    \]
\end{snippettheorem}

\begin{snippetproof}{polynomial-remainder-proof}{polynomial-remainder-theorem}{Remainder theorem}
    The polynomial \(x-a\) is monic, so polynomial division yields
    \[
        f(x)=(x-a)q(x)+r
    \]
    for a constant remainder \(r\in A\). Evaluating at \(a\) gives
    \(f(a)=r\).
\end{snippetproof}

\begin{snippetcorollary}{remainder-theorem-polynomial-root}{Root and linear factor}
    Let \(A\) be a commutative \ring, \(f\in A[x]\), and \(a\in A\).
    Then
    \[
        f(a)=0_A
        \ifandonlyif
        f(x)=(x-a)q(x)
    \]
    for some \(q\in A[x]\).
\end{snippetcorollary}

\begin{snippetdefinition}{polynomial-root-multiplicity-definition}{Polynomial root multiplicity}
    Let \(A\) be a commutative \ring, let \(f\in A[x]\) be nonzero, and
    let \(a\in A\) be a root of \(f\). The \emph{multiplicity} of \(a\)
    is the largest \(m\in\naturalnumbers\difference\{0\}\) such that
    \[
        (x-a)^m\divides f.
    \]
\end{snippetdefinition}

\begin{snippetdefinition}{algebraic-equation-definition}{Algebraic equation}
    Let \(A\) be a commutative \ring. An equation in one variable is
    \emph{algebraic over \(A\)} if it has the form
    \[
        f(x)=0_A
    \]
    for some nonzero \(f\in A[x]\).
\end{snippetdefinition}

\begin{snippettheorem}{solutions-of-algebraic-equation-theorem}{Bound on the number of roots}
    Let \(A\) be an \integraldomain. A nonzero \polynomial
    \(f\in A[x]\) has at most \(\polynomialdeg f\) distinct roots in
    \(A\).
\end{snippettheorem}

\begin{snippetproof}{solutions-of-algebraic-equation-proof}{solutions-of-algebraic-equation-theorem}{Bound on the number of roots}
    Let \(a_1,\ldots,a_t\) be distinct roots of \(f\). The root-factor
    criterion gives \(f=(x-a_1)q_1\). Since
    \[
        0=f(a_2)=(a_2-a_1)q_1(a_2)
    \]
    and \(A\) is an \integraldomain with \(a_2-a_1\neq0_A\), one has
    \(q_1(a_2)=0_A\). Repeating the argument yields
    \[
        f=(x-a_1)\cdots(x-a_t)q_t.
    \]
    Comparing degrees gives \(t\leq\polynomialdeg f\).
\end{snippetproof}

\section{Rational functions}

\begin{snippetdefinition}{rational-function-definition}{Rational function}
    Let \(K\) be a \field. The \emph{field of rational functions} in
    one indeterminate over \(K\) is the fraction field
    \[
        K(x)=\quotientfield(K[x]).
    \]
    Its elements are equivalence classes of quotients
    \[
        \frac{P(x)}{Q(x)},
        \qquad
        P,Q\in K[x],
        \quad
        Q\neq0.
    \]
\end{snippetdefinition}
