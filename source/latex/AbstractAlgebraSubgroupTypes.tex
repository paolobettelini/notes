\section{The centralizer subgroup}

\begin{snippetdefinition}{centralizer-group-definition}{The centralizer subgroup}
    Let \(G\) be a \group and let \(X\subseteq G\). The \emph{centralizer}
    of \(X\) in \(G\) is
    \[
        \mathrm{C}_G(X) \triangleq \{
            g \in G \suchthat \forall x \in X, gx=xg
        \}
    \]
    If \(x\in G\), write
    \(\mathrm{C}_G(x)\triangleq\mathrm{C}_G(\{x\})\).
\end{snippetdefinition}

\begin{plainhtml}
    The centralizer of one element contains at least the element itself, as it
    commutes with itself.
\end{plainhtml}

\begin{snippettheorem}{centralizer-of-subgroup-is-subgroup}{}
    Let \(G\) be a \group and let \(X\subseteq G\). Then,
    \(\groupcentralizer_G(X) \subgroupleq G\).
\end{snippettheorem}

\begin{snippetproof}{centralizer-of-subgroup-is-subgroup-proof}{centralizer-of-subgroup-is-subgroup}{Centralizer of subgroup is subgroup}
    The identity commutes with every element of \(X\), so the centralizer is
    non-empty. If \(a,b\in\groupcentralizer_G(X)\) and \(x\in X\), then
    \[
        (ab^{-1})x=a(b^{-1}x)=a(xb^{-1})=(ax)b^{-1}=x(ab^{-1}).
    \]
    Thus \(ab^{-1}\in\groupcentralizer_G(X)\), and the one-step subgroup
    test applies.
\end{snippetproof}

\begin{snippetproposition}{centralizer-subset-properties}{Centralizers of subsets}
    Let \(G\) be a \group and let \(X,Y\subseteq G\). Then:
    \begin{enumerate}
        \item if \(X\subseteq Y\), then
            \(\groupcentralizer_G(Y)\subgroupleq\groupcentralizer_G(X)\);
        \item \(\groupcentralizer_G(X)=
            \displaystyle\bigcap_{x\in X}\groupcentralizer_G(x)\);
        \item \(\groupcentralizer_G(X\union Y)=
            \groupcentralizer_G(X)\intersection\groupcentralizer_G(Y)\).
    \end{enumerate}
\end{snippetproposition}

\begin{snippetproof}{centralizer-subset-properties-proof}{centralizer-subset-properties}{Centralizers of subsets}
    Each assertion follows directly from the requirement that an element
    commute with every member of the indicated subset.
\end{snippetproof}

\section{Center of a group}

\begin{snippetdefinition}{center-of-group-definition}{Center of a group}
    Let \(G\) be a \group. The \emph{center} of \(G\) is
    \[
        \mathrm{Z}(G) \triangleq \mathrm{C}_G(G).
    \]
\end{snippetdefinition}

\begin{snippet}{center-of-group-condition-alternative}
    The condition \(gx=xg\) is also sometimes expressed as \(gxg^{-1} = x\).
\end{snippet}

\begin{snippettheorem}{center-of-group-is-subgroup}{}
    Let \(G\) be a \group, then \(\groupcenter(G) \subgroupleq G\).
\end{snippettheorem}

\begin{snippetproof}{center-of-group-is-subgroup-proof}{center-of-group-is-subgroup}{}
    Assume \(a, b \in \groupcenter(G)\), so \(a = gag^{-1}\) and
    \(b = gbg^{-1}\) for any \(g \in G\). \\
    We want to show \(ab^{-1} \in \groupcenter(G)\).
    \(ab^{-1} = (gag^{-1}){(gbg^{-1})}^{-1} = gag^{-1}gb^{-1}g^{-1}
    = g ab^{-1} g^{-1}\) which is precisely the requirement to be in \(\groupcenter(G)\).
\end{snippetproof}

\begin{snippetproposition}{generated-group-abelian-iff-subset-is}{Abelian group check}
    Let \(G=\gengrp{X}\) be a \group generated by \(X\subseteq G\).
    Then \(G\) is \abeliangroup[abelian] \ifandonlyif the elements of \(X\)
    commute pairwise.
\end{snippetproposition}

\begin{snippetproof}{generated-group-abelian-iff-subset-is-proof}{generated-group-abelian-iff-subset-is}{Abelian group check}
    \begin{iffproof}
        \item If \(G\) is abelian, every pair of elements of \(X\) commutes.
        \item Suppose the elements of \(X\) commute pairwise. For every
            \(x\in X\), the subgroup \(\groupcentralizer_G(x)\) contains
            \(X\), hence contains \(\gengrp{X}=G\). Thus every \(x\in X\)
            lies in \(\groupcenter(G)\). Since the center is a subgroup
            containing \(X\), it contains \(\gengrp{X}=G\). Therefore
            \(\groupcenter(G)=G\), and \(G\) is abelian.
    \end{iffproof}
\end{snippetproof}
