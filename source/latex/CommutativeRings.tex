\section{Fields and ideals}

\begin{snippetproposition}{field-ideals-trivial}{Ideals of a field}
    Let \(K\) be a \field. Its only \ideal[ideals] are
    \[
        \{0_K\}
        \qquad\text{and}\qquad
        K.
    \]
\end{snippetproposition}

\begin{snippetproof}{field-ideals-trivial-proof}{field-ideals-trivial}{Ideals of a field}
    Let \(I\idealin K\) be nonzero, and choose
    \(\alpha\in I\difference\{0_K\}\). Since \(\alpha\) is invertible and
    \(I\) is an \ideal,
    \[
        1_K=\alpha^{-1}\alpha\in I.
    \]
    Therefore \(\beta=\beta1_K\in I\) for every \(\beta\in K\), so
    \(I=K\).
\end{snippetproof}

\begin{snippetcorollary}{field-ring-homomorphism-injective}{Ring homomorphisms from fields are injective}
    Let \(K\) be a \field, let \(A\) be a nontrivial \ring, and let
    \(\varphi\colon K\fromto A\) be a \ringhomomorphism. Then \(\varphi\)
    is \injective.
\end{snippetcorollary}

\begin{snippetproof}{field-ring-homomorphism-injective-proof}{field-ring-homomorphism-injective}{Ring homomorphisms from fields are injective}
    The \ideal \(\ringker\varphi\) is either \(\{0_K\}\) or \(K\). The
    second possibility would imply
    \[
        1_A=\varphi(1_K)=0_A,
    \]
    contrary to the assumption that \(A\) is nontrivial. Hence
    \(\ringker\varphi=\{0_K\}\), and \(\varphi\) is \injective.
\end{snippetproof}

\begin{snippet}{field-ring-homomorphism-trivial-codomain-exception}
    The nontrivial-codomain hypothesis is necessary: if \(A=\{0_A\}\),
    the unique unital \ringhomomorphism \(K\fromto A\) is not
    \injective.
\end{snippet}

\section{Constructing the field of fractions}

\begin{snippetdefinition}{field-of-fractions-equivalence-definition}{Equivalence of formal fractions}
    Let \(A\) be an \integraldomain. On the \set
    \(A\times(A\difference\{0_A\})\), define
    \[
        (a_1,b_1)\sim(a_2,b_2)
        \ifandonlyif
        a_1b_2=a_2b_1.
    \]
\end{snippetdefinition}

\begin{snippetproposition}{field-of-fractions-equivalence-relation}{Formal-fraction equivalence is an equivalence relation}
    Let \(A\) be an \integraldomain. The relation
    \[
        (a_1,b_1)\sim(a_2,b_2)
        \ifandonlyif
        a_1b_2=a_2b_1
    \]
    is an \equivrelation on \(A\times(A\difference\{0_A\})\).
\end{snippetproposition}

\begin{snippetproof}{field-of-fractions-equivalence-relation-proof}{field-of-fractions-equivalence-relation}{Formal-fraction equivalence is an equivalence relation}
    Reflexivity and symmetry follow directly from equality. For
    transitivity, suppose
    \[
        a_1b_2=a_2b_1,
        \qquad
        a_2b_3=a_3b_2.
    \]
    Multiplying the first equality by \(b_3\), the second by \(b_1\), and
    using commutativity gives
    \[
        b_2(a_1b_3-a_3b_1)=0_A.
    \]
    Since \(b_2\neq0_A\) and \(A\) has no zero divisors,
    \(a_1b_3=a_3b_1\). Thus \((a_1,b_1)\sim(a_3,b_3)\).
\end{snippetproof}

\begin{snippetdefinition}{field-of-fractions-definition}{Field of fractions}
    Let \(A\) be an \integraldomain. Its \emph{field of fractions}, or
    \emph{quotient field}, is
    \[
        \quotientfield(A)
        =
        \big(A\times(A\difference\{0_A\})\big)/{\sim}.
    \]
    The class of \((a,b)\) is denoted by
    \[
        \frac{a}{b}=[(a,b)]_\sim.
    \]
    Addition and multiplication are defined by
    \[
        \frac{a_1}{b_1}+\frac{a_2}{b_2}
        =
        \frac{a_1b_2+a_2b_1}{b_1b_2},
        \qquad
        \frac{a_1}{b_1}\frac{a_2}{b_2}
        =
        \frac{a_1a_2}{b_1b_2}.
    \]
\end{snippetdefinition}

\begin{snippetproposition}{field-of-fractions-is-field}{The field of fractions is a field}
    Let \(A\) be an \integraldomain. The operations in the definition of
    \(\quotientfield(A)\) are well-defined and make
    \(\quotientfield(A)\) into a \field.
\end{snippetproposition}

\begin{snippetproof}{field-of-fractions-is-field-proof}{field-of-fractions-is-field}{The field of fractions is a field}
    Since \(A\) is an \integraldomain, products of nonzero denominators
    are nonzero, so the displayed operations always produce formal
    fractions.

    Suppose
    \[
        \frac{a_1}{b_1}=\frac{a_1'}{b_1'},
        \qquad
        \frac{a_2}{b_2}=\frac{a_2'}{b_2'}.
    \]
    Thus \(a_1b_1'=a_1'b_1\) and \(a_2b_2'=a_2'b_2\). Multiplying these
    equalities proves
    \[
        (a_1a_2)(b_1'b_2')=(a_1'a_2')(b_1b_2),
    \]
    so multiplication is independent of the representatives. Moreover,
    \begin{align*}
        &(a_1b_2+a_2b_1)b_1'b_2'\\
        &\qquad
        =a_1b_1'b_2b_2'+a_2b_2'b_1b_1'\\
        &\qquad
        =(a_1'b_2'+a_2'b_1')b_1b_2,
    \end{align*}
    so addition is also well-defined.

    The identities and additive inverse are
    \[
        0_{\quotientfield(A)}=\frac{0_A}{1_A},
        \qquad
        1_{\quotientfield(A)}=\frac{1_A}{1_A},
        \qquad
        -\frac{a}{b}=\frac{-a}{b}.
    \]
    The ring laws follow from those of \(A\). Finally,
    \[
        \frac{a}{b}=0_{\quotientfield(A)}
        \ifandonlyif a=0_A.
    \]
    Hence every nonzero fraction has inverse
    \[
        \left(\frac{a}{b}\right)^{-1}=\frac{b}{a}.
    \]
    Therefore \(\quotientfield(A)\) is a \field.
\end{snippetproof}

\section{Canonical embedding and examples}

\begin{snippetproposition}{canonical-field-of-fractions-embedding}{Canonical embedding into the field of fractions}
    Let \(A\) be an \integraldomain. The \function
    \[
        \iota\colon A\fromto\quotientfield(A),
        \qquad
        \iota(a)=\frac{a}{1_A},
    \]
    is an \injective \ringhomomorphism.
\end{snippetproposition}

\begin{snippetproof}{canonical-field-of-fractions-embedding-proof}{canonical-field-of-fractions-embedding}{Canonical embedding into the field of fractions}
    Direct computation gives
    \[
        \iota(a+b)=\iota(a)+\iota(b),
        \qquad
        \iota(ab)=\iota(a)\iota(b),
        \qquad
        \iota(1_A)=1_{\quotientfield(A)}.
    \]
    If \(\iota(a)=\iota(b)\), then
    \(a1_A=b1_A\), and hence \(a=b\). Thus \(\iota\) is \injective.
\end{snippetproof}

\begin{snippet}{field-of-fractions-generated-by-inverses}
    Identifying an \integraldomain \(A\) with its image under the canonical
    embedding, every element of \(\quotientfield(A)\) has the form
    \[
        \frac{a}{b}
        =
        \frac{a}{1_A}
        \left(\frac{b}{1_A}\right)^{-1}.
    \]
    Thus the field of fractions is obtained by adjoining inverses of all
    nonzero elements of \(A\).
\end{snippet}

\begin{snippetexample}{integers-field-of-fractions-example}{Fractions of the integers}
    The usual interpretation of a formal fraction gives a
    \ringisomorphism
    \[
        \quotientfield(\integers)\ringisomorphic\rationalnumbers.
    \]
\end{snippetexample}

\begin{snippetexample}{polynomial-field-of-fractions-example}{Rational functions}
    Let \(K\) be a \field. Then
    \[
        \quotientfield(K[x])=K(x),
    \]
    where
    \[
        K(x)
        =
        \left\{\frac{f(x)}{g(x)}
        \suchthat f,g\in K[x],\ g\neq0\right\}
    \]
    is the \field of rational functions in one variable.
\end{snippetexample}

\section{Universal property}

\begin{snippetproposition}{universal-property-field-fraction}{Universal property of the field of fractions}
    Let \(A\) be an \integraldomain, let \(L\) be a \field, and let
    \(\varphi\colon A\fromto L\) be an \injective \ringhomomorphism.
    There exists a unique \ringhomomorphism
    \[
        \widetilde{\varphi}\colon\quotientfield(A)\fromto L
    \]
    such that
    \[
        \varphi=\widetilde{\varphi}\circ\iota,
    \]
    where \(\iota\colon A\fromto\quotientfield(A)\) is the canonical
    embedding. Equivalently, the diagram
    \[
        \begin{tikzcd}
            A \arrow[r, "\iota", hook]
              \arrow[dr, "\varphi"']
                & \quotientfield(A)
                  \arrow[d, dashed, "\widetilde{\varphi}"] \\
                & L
        \end{tikzcd}
    \]
    commutes.
\end{snippetproposition}

\begin{snippetproof}{universal-property-field-fraction-proof}{universal-property-field-fraction}{Universal property of the field of fractions}
    Define
    \[
        \widetilde{\varphi}\left(\frac{a}{b}\right)
        =\varphi(a)\varphi(b)^{-1}.
    \]
    This is meaningful because \(b\neq0_A\) and injectivity gives
    \(\varphi(b)\neq0_L\). If \(a'/b'=a/b\), then \(a'b=ab'\), so
    \[
        \varphi(a')\varphi(b)
        =
        \varphi(a)\varphi(b').
    \]
    Multiplying by the inverses of \(\varphi(b)\) and \(\varphi(b')\)
    proves that the definition is independent of representatives.

    The formulas for addition and multiplication of fractions give
    \[
        \widetilde{\varphi}(u+v)
        =\widetilde{\varphi}(u)+\widetilde{\varphi}(v),
        \qquad
        \widetilde{\varphi}(uv)
        =\widetilde{\varphi}(u)\widetilde{\varphi}(v),
    \]
    and
    \(\widetilde{\varphi}(1_{\quotientfield(A)})=1_L\). Moreover,
    \[
        \widetilde{\varphi}(\iota(a))
        =
        \widetilde{\varphi}\left(\frac{a}{1_A}\right)
        =\varphi(a).
    \]

    If \(\psi\) is another such \ringhomomorphism, then
    \begin{align*}
        \psi\left(\frac{a}{b}\right)
        &=
        \psi\left(
            \frac{a}{1_A}
            \left(\frac{b}{1_A}\right)^{-1}
        \right)\\
        &=
        \varphi(a)\varphi(b)^{-1}
        =
        \widetilde{\varphi}\left(\frac{a}{b}\right).
    \end{align*}
    Hence \(\psi=\widetilde{\varphi}\).
\end{snippetproof}

\begin{snippetcorollary}{corollary-placeholder}{Uniqueness of the field of fractions}
    Let \(A\) be an \integraldomain. The field \(\quotientfield(A)\),
    together with its canonical embedding of \(A\), is unique up to a unique
    isomorphism compatible with that embedding.
\end{snippetcorollary}

\begin{snippetproof}{corollary-placeholder-proof}{corollary-placeholder}{Uniqueness of the field of fractions}
    Let \(K'\) be another \field with an embedding
    \(i'\colon A\fromto K'\) satisfying the same universal property.
    Applying the two universal properties gives \ringhomomorphism[ring
    homomorphisms]
    \[
        \Phi\colon\quotientfield(A)\fromto K',
        \qquad
        \Psi\colon K'\fromto\quotientfield(A)
    \]
    compatible with the embeddings. Both \(\Psi\circ\Phi\) and the
    identity extend the canonical embedding of \(A\); uniqueness forces
    them to coincide. Similarly \(\Phi\circ\Psi\) is the identity on
    \(K'\). Thus \(\Phi\) is the required unique \ringisomorphism.
\end{snippetproof}
