\section{Sigma algebras}

\begin{snippet}{sigma-algebra-motivation}
    A measure is generally not defined on every subset of a set \(X\).
    Instead, one selects a family of subsets that is stable under the
    countable set operations used in analysis. The letter \(\sigma\) emphasizes
    closure under countable unions.
\end{snippet}

\begin{snippetdefinition}{sigma-algebra-definition}{\(\sigma\)-algebra}
    Let \(X\) be a \set. A \emph{\(\sigma\)-algebra} on \(X\) is a family
    \(\Sigma\subseteq\powerset(X)\) such that
    \begin{enumerate}
        \item \(X\in\Sigma\);
        \item if \(A\in\Sigma\), then \(X\difference A\in\Sigma\);
        \item if \(A_n\in\Sigma\) for every \(n\in\naturalnumbers\), then
        \[
            \bigcup_{n=1}^{\infty}A_n\in\Sigma.
        \]
    \end{enumerate}
    The elements of \(\Sigma\) are called \emph{measurable sets}, or more
    specifically \emph{\(\Sigma\)-measurable sets}.
\end{snippetdefinition}

\begin{snippetcorollary}{empty-set-is-in-sigma-algebra}{The empty set is measurable}
    Let \(\Sigma\) be a \sigmaalgebra on \(X\). Then \(\emptyset\in\Sigma\).
\end{snippetcorollary}

\begin{snippetcorollary}{sigma-algebra-closed-under-intersection}{Closure properties of a \(\sigma\)-algebra}
    Let \(\Sigma\) be a \sigmaalgebra on \(X\). Then \(\Sigma\) is closed
    under countable intersections, set differences, and symmetric differences.
    In particular, if \(A_n\in\Sigma\) for every \(n\), and \(A,B\in\Sigma\), then
    \[
        \bigcap_{n=1}^{\infty}A_n\in\Sigma,
        \qquad
        A\difference B\in\Sigma,
        \qquad
        A\mathbin{\triangle}B\in\Sigma.
    \]
\end{snippetcorollary}

\begin{snippetproof}{sigma-algebra-closed-under-intersection-proof}{sigma-algebra-closed-under-intersection}{Closure properties of a \(\sigma\)-algebra}
    The empty set is the complement of \(X\). De Morgan's laws give
    \[
        \bigcap_{n=1}^{\infty}A_n
        =X\difference\bigcup_{n=1}^{\infty}(X\difference A_n).
    \]
    Moreover,
    \[
        A\difference B=A\intersection(X\difference B),
        \qquad
        A\mathbin{\triangle}B=(A\difference B)\union(B\difference A).
    \]
    Each set on the right belongs to \(\Sigma\).
\end{snippetproof}

\begin{snippetexample}{trivial-sigma-algebras-example}{Basic \(\sigma\)-algebras}
    For every \set \(X\), both
    \[
        \{\emptyset,X\}
        \qquad\text{and}\qquad
        \powerset(X)
    \]
    are \(\sigma\)-algebras on \(X\). They are respectively the smallest and
    largest \(\sigma\)-algebras on \(X\).
\end{snippetexample}

\section{Generated sigma algebras}

\begin{snippetproposition}{intersection-of-sigma-algebras-proposition}{Intersection of \(\sigma\)-algebras}
    Let \(X\) be a \set and let \(\{\Sigma_i\}_{i\in I}\) be a nonempty family
    of \(\sigma\)-algebras on \(X\). Then
    \[
        \bigcap_{i\in I}\Sigma_i
    \]
    is a \(\sigma\)-algebra on \(X\).
\end{snippetproposition}

\begin{snippetproof}{intersection-of-sigma-algebras-proposition-proof}{intersection-of-sigma-algebras-proposition}{Intersection of \(\sigma\)-algebras}
    Every \(\Sigma_i\) contains \(X\), so their intersection contains \(X\).
    If a set belongs to every \(\Sigma_i\), then so does its complement. If
    every set in a sequence belongs to every \(\Sigma_i\), then its union also
    belongs to every \(\Sigma_i\). Thus the intersection satisfies all three
    defining properties of a \(\sigma\)-algebra.
\end{snippetproof}

\begin{snippetdefinition}{smallest-sigma-algebra-definition}{Generated \(\sigma\)-algebra}
    Let \(X\) be a \set and let \(\mathcal S\subseteq\powerset(X)\). The
    \emph{\(\sigma\)-algebra generated by \(\mathcal S\)} is
    \[
        \sigma(\mathcal S)
        =\bigcap\{\Sigma\subseteq\powerset(X)
        \suchthat \Sigma\text{ is a \(\sigma\)-algebra and }
        \mathcal S\subseteq\Sigma\}.
    \]
    It is the smallest \(\sigma\)-algebra on \(X\) containing \(\mathcal S\).
\end{snippetdefinition}

\begin{snippetexample}{countable-cocountable-sigma-algebra-example}{The countable--cocountable \(\sigma\)-algebra}
    Let \(X\) be an uncountable \set. The family
    \[
        \Sigma=\{A\subseteq X\suchthat A\text{ is countable or }
        X\difference A\text{ is countable}\}
    \]
    is a \(\sigma\)-algebra. It is generated by the finite subsets of \(X\).
    By contrast, the family of all subsets containing a fixed point of \(X\)
    is not a \(\sigma\)-algebra because it is not closed under complements.
\end{snippetexample}

\section{Measurable spaces}

\begin{snippetdefinition}{measurable-space-definition}{Measurable space}
    A \emph{measurable space} is a pair \((X,\Sigma)\), where \(X\) is a
    \set and \(\Sigma\) is a \sigmaalgebra on \(X\).
\end{snippetdefinition}
