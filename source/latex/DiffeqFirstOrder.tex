\section{First-order equations}

\begin{snippetdefinition}{diffeq-first-order-normal-definition}{First-order normal form}
    Let \(\Omega\subseteq\realnumbers^2\) be open and
    \(f\colon\Omega\fromto\realnumbers\) continuous. A first-order equation
    in normal form is \(y'=f(x,y)\). A solution is a differentiable
    \function \(y\colon I\fromto\realnumbers\), on an open interval \(I\),
    such that \((x,y(x))\in\Omega\) and \(y'(x)=f(x,y(x))\) for all \(x\in I\).
    Continuity of \(f\) then implies \(y\in\continuityclass^1(I)\).
\end{snippetdefinition}

\begin{snippettheorem}{diffeq-local-existence-uniqueness}{Local existence and uniqueness}
    Let \(\Omega\subseteq\realnumbers^2\) be open and suppose
    \(f,\partial f/\partial y\colon\Omega\fromto\realnumbers\) are continuous.
    For every \((x_0,y_0)\in\Omega\), there is an open interval \(I\) containing
    \(x_0\) on which \(y'=f(x,y)\), \(y(x_0)=y_0\), has a unique solution.
    Any two solutions with these initial data agree on the intersection of their domains.
\end{snippettheorem}

\begin{snippetproof}{diffeq-local-existence-uniqueness-proof}{diffeq-local-existence-uniqueness}{Local existence and uniqueness}
    Around the initial point choose a compact rectangle contained in \(\Omega\).
    Continuity makes \(\partial_y f\) bounded there, and the mean value theorem
    gives a state Lipschitz bound uniform in time.
    The \snippetref[diffeq-cauchy-lipschitz-cylinder][Cauchy--Lipschitz theorem]
    gives existence and uniqueness on a neighborhood of the initial time.
    Agreement throughout any common interval follows from the open-and-closed
    argument for the set where the two solutions coincide.
\end{snippetproof}

\begin{plainhtml}
    This local theorem is stated here to justify the explicit methods.
    The subsequent pages on existence, continuation and continuous dependence
    give the vector-valued theory and its proofs.
\end{plainhtml}

\begin{snippetcorollary}{diffeq-solutions-noncrossing}{Nonintersection of solution graphs}
    Suppose \(\Omega\subseteq\realnumbers^2\) is open and \(f,\partial_y f\)
    are continuous on \(\Omega\). If two solutions of \(y'=f(x,y)\)
    meet at a point, they agree on the intersection of their interval domains.
    In particular, distinct maximal solutions cannot intersect.
\end{snippetcorollary}

\begin{snippetproof}{diffeq-solutions-noncrossing-proof}{diffeq-solutions-noncrossing}{Nonintersection of solution graphs}
    Let \(J\) be the intersection of the two domains. The \set of points of
    \(J\) where the solutions agree is nonempty and relatively closed by continuity.
    Local uniqueness makes it relatively open. Since \(J\) is an interval,
    they agree throughout \(J\). If both are maximal, their common values allow
    them to be joined on the union of their domains; maximality then forces
    equality of both domains and solutions.
\end{snippetproof}

\begin{snippetexample}{diffeq-discontinuous-no-solution}{Failure of existence}
    Let
    \[
        f(x)=\operatorname{sign}(x)=
        \begin{cases}1,&x>0,\\0,&x=0,\\-1,&x<0.\end{cases}
    \]
    The equation \(y'=f(x)\) has no solution on an open interval containing \(0\).
    Indeed, on its positive and negative parts a solution would respectively be
    \(x+C_+\) and \(-x+C_-\). Continuity at zero forces
    \(C_+=C_-\), but the left and right derivatives at zero would be \(-1\) and \(1\).
\end{snippetexample}

\section{Geometric and numerical interpretation}

\begin{snippetdefinition}{slope-field-definition}{Slope field}
    For \(y'=f(x,y)\) on \(\Omega\subseteq\realnumbers^2\), the
    \emph{slope field} assigns to every \((x,y)\in\Omega\) the line
    with direction vector \((1,f(x,y))\). The graph of each solution
    is tangent to these directions at its points.
\end{snippetdefinition}

\begin{snippetdefinition}{euler-method}{Explicit Euler method}
    For the initial value problem \(y'=f(x,y)\), \(y(x_0)=y_0\),
    choose a step size \(h>0\). The \emph{explicit Euler method} defines
    \[
        x_j=x_0+jh,\qquad Y_0=y_0,\qquad
        Y_{j+1}=Y_j+h f(x_j,Y_j),
    \]
    for as long as the required points lie in the domain of \(f\).
\end{snippetdefinition}

\begin{snippet}{diffeq-euler-interpretation}
    Each Euler step follows the slope field at the current approximate point.
    The polygon through \((x_j,Y_j)\) approximates a solution.
    Convergence as \(h\to0\) requires suitable regularity and stability assumptions;
    it is not part of the definition and does not hold for arbitrary right-hand sides.
\end{snippet}
