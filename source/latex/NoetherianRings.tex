\section{Noetherian rings}

\begin{snippetdefinition}{noetherian-ring-definition}{Noetherian ring}
    A \commutativering \(A\) is \emph{Noetherian} if every ascending chain
    of \ideal[ideals]
    \[
        I_1\subseteq I_2\subseteq I_3\subseteq\cdots
    \]
    stabilizes: there exists \(m\) such that
    \[
        I_m=I_{m+1}=I_{m+2}=\cdots.
    \]
\end{snippetdefinition}

\begin{snippettheorem}{noetherian-ring-equivalence-theorem}{Characterizations of Noetherian rings}
    Let \(A\) be a \commutativering. The following conditions are
    equivalent:
    \begin{enumerate}
        \item \(A\) is a \noetherianring;
        \item every \ideal of \(A\) is finitely generated;
        \item every nonempty \set of \ideal[ideals] of \(A\) has a maximal
        element with respect to inclusion.
    \end{enumerate}
\end{snippettheorem}

\begin{snippetproof}{noetherian-ring-equivalence-theorem-proof}{noetherian-ring-equivalence-theorem}{Characterizations of Noetherian rings}
    Suppose first that \(A\) is a \noetherianring. If a nonempty family
    \(\mathcal{S}\) of \ideal[ideals] had no maximal element, one could
    inductively choose
    \[
        I_1\subsetneq I_2\subsetneq I_3\subsetneq\cdots,
        \qquad
        I_i\in\mathcal{S},
    \]
    contradicting stabilization. Thus \((1)\) implies \((3)\).

    Suppose \((3)\) holds, and let \(I\idealin A\). Consider the nonempty
    \set
    \[
        \mathcal{S}
        =
        \{J\idealin A\suchthat
        J\subseteq I\text{ and }J\text{ is finitely generated}\}.
    \]
    Let \(J\) be maximal in \(\mathcal{S}\). If \(J\neq I\), choose
    \(y\in I\difference J\). Then \(J+(y)\) is a finitely generated
    \ideal contained in \(I\) and strictly containing \(J\), a
    contradiction. Hence \(J=I\), proving \((2)\).

    Finally, suppose every \ideal is finitely generated, and let
    \[
        I_1\subseteq I_2\subseteq I_3\subseteq\cdots
    \]
    be an ascending chain. Its union
    \[
        J=\bigcup_{i\geq1}I_i
    \]
    is an \ideal because the chain is totally ordered by inclusion. Write
    \[
        J=(x_1,\ldots,x_n).
    \]
    For each \(k\), choose \(i_k\) with \(x_k\in I_{i_k}\), and set
    \(m=\max\{i_1,\ldots,i_n\}\). Then every generator of \(J\) belongs to
    \(I_m\), so \(J\subseteq I_m\). Since \(I_m\subseteq J\), one has
    \(J=I_m\), and consequently \(I_h=I_m\) for every \(h\geq m\).
\end{snippetproof}

\section{Hilbert's basis theorem}

\begin{snippettheorem}{hilbert-basis-theorem}{Hilbert's basis theorem}
    If \(A\) is a \noetherianring, then \(A[x]\) is a
    \noetherianring.
\end{snippettheorem}

\begin{snippetproof}{hilbert-basis-theorem-proof}{hilbert-basis-theorem}{Hilbert's basis theorem}
    By the preceding characterization, it suffices to prove that every
    \ideal \(J\idealin A[x]\) is finitely generated. The assertion is
    immediate for \(J=\{0\}\), so suppose \(J\neq\{0\}\).

    For each \(i\geq0\), let \(I_i\) be the \set consisting of \(0_A\)
    together with all leading coefficients of degree-\(i\)
    \polynomial[polynomials] in \(J\). Then \(I_i\idealin A\). Indeed,
    subtraction of two degree-\(i\) polynomials subtracts their leading
    coefficients, unless the leading terms cancel, in which case the
    difference of those coefficients is \(0_A\); multiplication by an
    element of \(A\) proves absorption.

    Multiplication by \(x\) shows that
    \[
        I_0\subseteq I_1\subseteq I_2\subseteq\cdots.
    \]
    Since \(A\) is a \noetherianring, there is \(m\) such that
    \[
        I_m=I_{m+1}=I_{m+2}=\cdots.
    \]
    Each \(I_i\) is finitely generated. For \(0\leq i\leq m\), choose
    nonzero generators \(a_{i,1},\ldots,a_{i,r_i}\) of \(I_i\), omitting
    the family when \(I_i=\{0_A\}\). For each generator choose
    \(f_{i,h}\in J\) of degree \(i\) with leading coefficient \(a_{i,h}\).
    Let \(J'\) be the \ideal of \(A[x]\) generated by this finite family.
    By construction, \(J'\subseteq J\).

    We prove \(J\subseteq J'\) by induction on degree. Let \(g\in J\) be
    nonzero of degree \(s\), with leading coefficient \(a\). If \(s\leq m\),
    write
    \[
        a=\sum_{h=1}^{r_s}c_ha_{s,h}.
    \]
    Then
    \[
        g-\sum_{h=1}^{r_s}c_hf_{s,h}
    \]
    belongs to \(J\) and has degree strictly less than \(s\). If \(s>m\),
    use \(I_s=I_m\) to write
    \[
        a=\sum_{h=1}^{r_m}c_ha_{m,h},
    \]
    and subtract
    \[
        \sum_{h=1}^{r_m}c_hx^{s-m}f_{m,h}.
    \]
    Again the result lies in \(J\) and has smaller degree. In either case,
    the induction hypothesis places the remainder in \(J'\), and the
    subtracted polynomial already belongs to \(J'\). Thus \(g\in J'\).

    Therefore \(J=J'\), so \(J\) is finitely generated. Hence \(A[x]\) is
    a \noetherianring.
\end{snippetproof}

\begin{snippetcorollary}{noetherian-polynomial-ring-corollary}{Polynomial rings in finitely many variables}
    If \(A\) is a \noetherianring, then
    \[
        A[x_1,\ldots,x_n]
    \]
    is a \noetherianring for every \(n\geq1\).
\end{snippetcorollary}

\begin{snippetproof}{noetherian-polynomial-ring-corollary-proof}{noetherian-polynomial-ring-corollary}{Polynomial rings in finitely many variables}
    Apply Hilbert's basis theorem repeatedly, using
    \[
        A[x_1,\ldots,x_n]
        =
        A[x_1,\ldots,x_{n-1}][x_n].
    \]
\end{snippetproof}
