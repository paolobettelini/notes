\section{Homomorphism and generators}

\begin{snippetproposition}{equal-maps-subgroup}{}
    Let \(\varphi\) and \(\psi\) be \grouphomomorphism[group homomorphisms]
    from \(G\) to \(H\). The subset \(S\) of \(G\)
    in which \(\varphi\) and \(\phi\) coincide is a \subgroup of \(G\)
\end{snippetproposition}

\begin{snippetproof}{equal-maps-subgroup-proof}{equal-maps-subgroup}{}
    \begin{enumerate}
        \item \(1_G \in S\): Indeed, \(\varphi(1_G) = 1_H = \psi(1_G) = 1_H\);
        \item if \(x\) and \(y\) are in \(S\), meaning \(\varphi(x) = \psi(x)\)
        and \(\varphi(y) = \psi(y)\), then
        \[
            \varphi(xy) = \varphi(x)\varphi(y)=\psi(x)\psi(y) = \psi(xy)
        \]
        meaning \(xy \in S\);
        \item if \(x\in S\), meaning \(\varphi(x) = \psi(x)\), then
        \[
            \varphi(x^{-1}) = {(\varphi(x))}^{-1} = \psi(x^{-1})    
        \]
        meaning \(x^{-1} \in S\).
    \end{enumerate}
\end{snippetproof}

\begin{snippetcorollary}{equal-maps-on-generated-subgroup}{}
    Let \(\varphi\) and \(\psi\) be \grouphomomorphism[group homomorphisms]
    from \(G\) to \(H\) and let \(X\) be a subset
    of \(G\) in which  \(\varphi\) and \(\phi\) coincide.
    Then, \(\varphi\) and \(\psi\) coincide in \(\gengrp{X}\).
    In particular, if \(G = \gengrp{X}\), then \(\varphi = \psi\).
\end{snippetcorollary}

\begin{plainhtml}
    If we know how the homomorphisms behave on the generators, we can reconstruct the entire homomorphism.
    This does not mean that we can randomly associate images to the generators, but sometimes we can.
\end{plainhtml}

\section{Homomorphism from cyclic groups}

\begin{snippet}{homomorphisms-from-cyclic-group-expl}
    Let \(G = \gengrp{g}\) be a \cyclicgroup. We want to define a \grouphomomorphism
    from \(G\) to \(H\).
    We know that \(\varphi \colon G \to H\)
    is defined by the information of \(\varphi(g)\).
    Since the elements of \(G\) are the powers of \(g\),
    we can say that \(\varphi(g^n) = {(\varphi(g))}^n\).
    If \(h\in H\) and we define \(\varphi(g) = h\),
    we cannot just say that \(\varphi(g^n) = h^n\). We need to make sure that if \(|G| =d\),
    the period of \(h\) divides \(d\).\\
    If \(d\) is infinite, this \function is well-defined (\(g^n = g^m \iff m=n\)). \\
    If \(d\) is finite, this \function might not be (\(g^n = g^m \iff m \equiv n \pmod{d}\))
    and thus \(m-n\) must be a multiple of the period of \(h\).
    If the \function is well-defined, it is also a \grouphomomorphism:
    two elements of \(G\) have form \(g^n\) and \(g^m\) for some \(n,m\in\integers\)
    and we have \(h^n = \varphi(g^n)\), \(h^m = \varphi(g^m)\)
    and \(\varphi(g^n g^m) = \varphi(g^{m+n})\).
\end{snippet}

\begin{snippettheorem}{infinite-ciclicgroup-homomorphism-theorem}{}
    Let \(G = \gengrp{g}\) be an infinite \cyclicgroup
    and \(H\) be a \group.
    Given \(h\in H\), there exists a unique
    \grouphomomorphism \(\varphi \colon G \to H\)
    such that \(\varphi(g) = h\).
\end{snippettheorem}

\begin{snippetproof}{infinite-ciclicgroup-homomorphism-theorem-proof}{infinite-ciclicgroup-homomorphism-theorem}{}
    Define \(\varphi(g^k)=h^k\) for \(k\in\integers\). Since the powers of
    \(g\) are all distinct, this is well-defined, and
    \(\varphi(g^{k+m})=h^{k+m}=h^kh^m\), so it is a homomorphism. Any
    homomorphism sending \(g\) to \(h\) must have these values, proving uniqueness.
\end{snippetproof}

\begin{snippettheorem}{finite-ciclicgroup-homomorphism-theorem}{}
    Let \(G = \gengrp{g}\) be a finite \cyclicgroup
    and \(H\) be a \group.
    Given \(h\in H\), there exists a \grouphomomorphism
    \(\varphi\colon G \to H\) such that \(\varphi(g) = h\)
    \ifandonlyif the period of \(h\) divides \(|G|\).
    In such case, \(\varphi\) is unique.
\end{snippettheorem}

\begin{snippetproof}{finite-ciclicgroup-homomorphism-theorem-proof}{finite-ciclicgroup-homomorphism-theorem}{}
    Let \(|G|=n\). If such a homomorphism exists, then
    \(h^n=\varphi(g^n)=1_H\), so the period of \(h\) divides \(n\).
    Conversely, under that divisibility condition define
    \(\varphi(g^k)=h^k\). If \(g^k=g^m\), then \(n\divides(k-m)\), and the
    period condition gives \(h^k=h^m\), so the map is well-defined. The
    homomorphism law and uniqueness follow as in the infinite case.
\end{snippetproof}

\begin{snippetcorollary}{cyclic-group-isomorphism}{Isomorphism between cyclic groups}
    Two \cyclicgroup[cyclic groups] are isomorphic
    \ifandonlyif \(|G| = |H|\).
\end{snippetcorollary}

\begin{snippetcorollary}{cyclic-group-classification}{Classification of cyclic groups}
    Every infinite \cyclicgroup is isomorphic to \((\integers,+)\), and every
    finite cyclic group of order \(n\) is isomorphic to
    \((\integers/n\integers,+)\).
\end{snippetcorollary}

\begin{snippetproof}{cyclic-group-classification-proof}{cyclic-group-classification}{Classification of cyclic groups}
    If \(G=\gengrp{g}\), the map \(k\mapsto g^k\) is a surjective
    homomorphism from \(\integers\) to \(G\). Its kernel is trivial when
    \(g\) has infinite order, and is \(n\integers\) when \(g\) has order
    \(n\). The first isomorphism theorem gives the two claims.
\end{snippetproof}

\begin{snippetproof}{cyclic-group-isomorphism-proof}{cyclic-group-isomorphism}{isomorphism between cyclic groups}
    \begin{iffproof}
        \item If \(G\) and \(H\) are isomorphic, then \(|G| = |H|\).
        \item If \(|G| = |H|\), let \(G = \gengrp{g}\)
        and \(H = \gengrp{h}\).
        If \(G\) and \(H\) are infinite we have a \grouphomomorphism \(\varphi \colon G \to H\)
        which sends \(g\) to \(h\) and a \grouphomomorphism \(\psi\colon H \to G\)
        that sends \(h\) to \(g\). Clearly, \(\varphi\) and \(\psi\)
        are inverses of eachtoher.
        If \(G\) and \(H\) are finite, we can do the same:
        there is a \group \(\varphi \colon G \to H\) which sends \(g\) in \(h\)
        (\(|h|\) divides \(|g|\)) and there exist a \grouphomomorphism
        \(\psi \colon H \to G\) which sends \(h\) in \(g\). The same goes for this case.
    \end{iffproof}
\end{snippetproof}

\subsection{Cyclic endomorphisms and automorphisms}

\begin{plainhtml}
    For a finite cyclic group, we have as many endomorphisms as there are elements of the group.
\end{plainhtml}

\begin{snippettheorem}{endomorphisms-automorphisms-cyclic-groups}{Endomorphisms and automorphisms of cyclic groups}
    Let \(G=\gengrp{g}\) be a \cyclicgroup.
    \begin{enumerate}
        \item If \(G\) is infinite, then
            \(\operatorname{End}(G)\) is isomorphic to the multiplicative
            monoid \((\integers,\cdot)\), and
            \(\operatorname{Aut}(G)\groupisomorphic C_2\).
        \item If \(|G|=n\), then
            \[
                \operatorname{End}(G)\cong(\integers/n\integers,\cdot),
                \qquad
                \operatorname{Aut}(G)\groupisomorphic
                {(\integers/n\integers)}^{\times}.
            \]
            In particular, \(|\operatorname{Aut}(G)|=\eulertotient(n)\).
    \end{enumerate}
\end{snippettheorem}

\begin{snippetproof}{endomorphisms-automorphisms-cyclic-groups-proof}{endomorphisms-automorphisms-cyclic-groups}{Endomorphisms and automorphisms of cyclic groups}
    Every endomorphism is uniquely determined by
    \(g\mapsto g^k\); denote it by \(\varphi_k\). Composition satisfies
    \(\varphi_k\varphi_m=\varphi_{km}\) under the right-composition
    convention. If \(G\) is infinite, distinct integers give distinct maps,
    and \(\varphi_k\) is invertible exactly when \(k=\pm1\).

    If \(|G|=n\), then \(\varphi_k=\varphi_m\) exactly when
    \(k\equiv m\pmod n\). Moreover, \(\varphi_k\) is an automorphism exactly
    when \(g^k\) has order \(n\), equivalently when \(\gcd(k,n)=1\).
    This proves both isomorphisms and the cardinality formula.
\end{snippetproof}

\begin{snippettheorem}{automorphism-group-cyclic-prime-order}{Automorphisms of a cyclic group of prime order}
    Let \(q\) be prime. Then
    \[
        \operatorname{Aut}(C_q)\groupisomorphic
        {(\integers/q\integers)}^{\times}
    \]
    is cyclic of order \(q-1\).
\end{snippettheorem}

\begin{snippetproof}{automorphism-group-cyclic-prime-order-proof}{automorphism-group-cyclic-prime-order}{Automorphisms of a cyclic group of prime order}
    The preceding theorem gives the displayed isomorphism and its order.
    Let \(m\) be the exponent of the finite abelian group
    \({(\integers/q\integers)}^{\times}\). Every nonzero residue is a root of
    \(X^m-1\) modulo \(q\). A nonzero polynomial of degree \(m\) over
    \(\integers/q\integers\) has at most \(m\) roots, so \(q-1\leq m\).
    On the other hand, Lagrange's theorem gives \(m\leq q-1\), hence
    \(m=q-1\). A finite abelian group contains an element whose order equals
    its exponent: for each prime divisor, choose an element realizing the
    largest corresponding prime power and multiply these commuting elements.
    Therefore the unit group contains an element of order \(q-1\) and is cyclic.
\end{snippetproof}
