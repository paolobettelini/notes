\section{Definitions}

\begin{snippetdefinition}{ring-definition}{Ring}
    A \emph{ring} is a \set \(A\) equipped with two
    \binoperation[binary operations], addition and multiplication, such that:
    \begin{enumerate}
        \item \((A,+)\) is an \abeliangroup, with identity \(0_A\);
        \item \((A,\cdot)\) is a \monoid, with identity \(1_A\);
        \item for every \(a,b,c\in A\),
        \[
            a(b+c)=ab+ac,
            \qquad
            (a+b)c=ac+bc.
        \]
    \end{enumerate}
    Ring homomorphisms in these notes are required to preserve \(1\).
\end{snippetdefinition}

\begin{snippetdefinition}{commutative-ring-definition}{Commutative ring}
    A \ring \(A\) is said to be \emph{commutative} if
    \[
        ab=ba
    \]
    for all \(a,b\in A\).
\end{snippetdefinition}

\begin{snippetproposition}{trivial-ring-characterization}{The trivial ring}
    Let \(A\) be a \ring. Then \(0_A=1_A\) \ifandonlyif
    \(A=\{0_A\}\).
\end{snippetproposition}

\begin{snippetproof}{trivial-ring-characterization-proof}{trivial-ring-characterization}{The trivial ring}
    If \(0_A=1_A\), then for every \(a\in A\),
    \[
        a=a1_A=a0_A=0_A.
    \]
    Hence \(A=\{0_A\}\). The converse is immediate.
\end{snippetproof}

\includesnpt{field-definition}

\begin{snippetdefinition}{division-ring-definition}{Division ring}
    A \emph{division ring} is a nontrivial \ring \(A\) in which every
    nonzero element has a two-sided multiplicative inverse. A division ring
    is also called a \emph{skew field}. It need not be commutative.
\end{snippetdefinition}

\section{Examples}

\begin{snippetexample}{integers-ring-example}{The ring of integers}
    The \set \(\integers\), with its usual addition and multiplication, is a
    \commutativering. Its only invertible elements are \(1\) and \(-1\).
\end{snippetexample}

\begin{snippetexample}{polynomial-ring-over-field-example}{A polynomial ring}
    Let \(K\) be a \field. The \set \(K[x]\) of \polynomial[polynomials]
    with coefficients in \(K\) is a \commutativering. Its invertible
    elements are precisely the nonzero constant polynomials, namely the
    elements of \(K^\times\).
\end{snippetexample}

\begin{snippetexample}{matrix-ring-example}{Matrix rings}
    Let \(K\) be a \field and \(n\geq1\). The \set
    \(\matrices_n(K)\) of \(n\times n\) \matrix[matrices], with matrix
    addition and multiplication, is a \ring. Its additive and multiplicative
    identities are the zero matrix \(0_n\) and the identity matrix \(I_n\).
    If \(n>1\), the ring is not commutative. Its invertible elements are
    precisely the matrices in \(\operatorname{GL}_n(K)\).
\end{snippetexample}

\begin{snippetexample}{standard-fields-example}{Standard fields}
    The number systems
    \[
        \rationalnumbers,\qquad \realnumbers,\qquad \complexnumbers
    \]
    are \field[fields]. If \(p\) is a \primen, then
    \(\integers/p\integers\) is also a \field.
\end{snippetexample}

\begin{snippetexample}{function-ring-example}{Rings of functions}
    Let \(A\) be a \ring and \(X\) a \set. The \set
    \[
        A^X=\{f\colon X\fromto A\}
    \]
    of all \function[functions] from \(X\) to \(A\) is a \ring under
    pointwise operations:
    \[
        (f+g)(x)=f(x)+g(x),
        \qquad
        (fg)(x)=f(x)g(x).
    \]
    Its identities are the constant \function[functions]
    \(0_{A^X}(x)=0_A\) and \(1_{A^X}(x)=1_A\).
\end{snippetexample}

\begin{snippetexample}{hamilton-quaternions-ring-example}{Hamilton's quaternions}
    The Hamilton quaternions form the real \vectorspace
    \[
        \mathbb{H}
        =\operatorname{span}_{\realnumbers}\{1,i,j,k\},
    \]
    with multiplication determined bilinearly by
    \[
        i^2=j^2=k^2=-1,
        \qquad
        ij=k,\quad jk=i,\quad ki=j,
    \]
    and
    \[
        ji=-k,\quad kj=-i,\quad ik=-j.
    \]
    Thus \(\mathbb{H}\) is not commutative. For
    \(q=a+bi+cj+dk\), define
    \[
        \overline q=a-bi-cj-dk.
    \]
    Then
    \[
        q\overline q=\overline q q=a^2+b^2+c^2+d^2,
    \]
    so every \(q\neq0\) has inverse
    \[
        q^{-1}
        =\frac{\overline q}{a^2+b^2+c^2+d^2}.
    \]
    Hence \(\mathbb{H}\) is a noncommutative division ring.
\end{snippetexample}

\section{Elementary properties}

\begin{snippetproposition}{ring-zero-multiplication}{Multiplication by zero}
    Let \(A\) be a \ring. For every \(a\in A\),
    \[
        a0_A=0_Aa=0_A.
    \]
\end{snippetproposition}

\begin{snippetproof}{ring-zero-multiplication-proof}{ring-zero-multiplication}{Multiplication by zero}
    Since \(0_A=0_A+0_A\),
    \[
        a0_A=a(0_A+0_A)=a0_A+a0_A.
    \]
    Cancellation in the additive \group gives \(a0_A=0_A\). The proof that
    \(0_Aa=0_A\) is analogous.
\end{snippetproof}

\begin{snippetproposition}{ring-inverse-addition-multiplication}{Multiplication by minus one}
    Let \(A\) be a \ring. For every \(a\in A\),
    \[
        a(-1_A)=(-1_A)a=-a.
    \]
\end{snippetproposition}

\begin{snippetproof}{ring-inverse-addition-multiplication-proof}{ring-inverse-addition-multiplication}{Multiplication by minus one}
    From \(1_A+(-1_A)=0_A\), distributivity gives
    \[
        a+a(-1_A)=a(1_A+(-1_A))=a0_A=0_A.
    \]
    Thus \(a(-1_A)\) is the additive inverse of \(a\). Multiplying the same
    equality on the right by \(a\) proves \((-1_A)a=-a\).
\end{snippetproof}

\begin{snippetproposition}{ring-multiplication-as-addition}{Integer multiples in a ring}
    Let \(A\) be a \ring, let \(a,b\in A\), and let
    \(n\in\naturalnumbers\). Writing
    \[
        na=\underbrace{a+\cdots+a}_{n\text{ times}},
    \]
    one has
    \[
        n(ab)=(na)b=a(nb).
    \]
    Moreover,
    \[
        (-a)b=a(-b)=-(ab),
        \qquad
        (-a)(-b)=ab.
    \]
\end{snippetproposition}

\begin{snippetproof}{ring-multiplication-as-addition-proof}{ring-multiplication-as-addition}{Integer multiples in a ring}
    The equality \(n(ab)=(na)b=a(nb)\) follows by induction on \(n\), using
    distributivity in the induction step. Since \(-a=(-1_A)a\),
    associativity and multiplication by \(-1_A\) yield
    \[
        (-a)b=(-1_A)(ab)=-(ab).
    \]
    The remaining identities follow in the same way.
\end{snippetproof}

\section{Zero divisors and cancellation}

\begin{snippetdefinition}{ring-zero-divisor-definition}{Zero divisor}
    Let \(A\) be a \ring and let \(a\in A\setminus\{0_A\}\). The element
    \(a\) is a \emph{left zero divisor} if there exists
    \(b\in A\setminus\{0_A\}\) such that \(ab=0_A\), and it is a
    \emph{right zero divisor} if there exists
    \(b\in A\setminus\{0_A\}\) such that \(ba=0_A\). A \emph{zero divisor}
    is an element that is a left or right zero divisor.
\end{snippetdefinition}

\begin{snippetdefinition}{integral-domain-definition}{Integral domain}
    An \emph{integral domain} is a nontrivial \commutativering with no
    nonzero zero divisors.
\end{snippetdefinition}

\begin{snippetexample}{matrix-ring-zero-divisors-example}{Zero divisors in a matrix ring}
    Let \(K\) be a \field. In \(\matrices_2(K)\), the nonzero matrices
    \[
        A=\begin{pmatrix}0&1\\0&0\end{pmatrix},
        \qquad
        B=\begin{pmatrix}1&0\\0&0\end{pmatrix}
    \]
    satisfy \(AB=0\), whereas
    \[
        BA=A\neq0.
    \]
    Thus left and right zero-divisor behavior can differ in a
    noncommutative \ring.
\end{snippetexample}

\begin{snippetproposition}{zero-divisors-are-not-invertible}{Zero divisors are not invertible}
    A zero divisor in a \ring is not invertible.
\end{snippetproposition}

\begin{snippetproof}{zero-divisors-are-not-invertible-proof}{zero-divisors-are-not-invertible}{Zero divisors are not invertible}
    Suppose that \(a\) is invertible and \(ab=0_A\). Multiplication on the
    left by \(a^{-1}\) gives \(b=0_A\), so \(a\) cannot be a left zero
    divisor. If \(ba=0_A\), multiplication on the right by \(a^{-1}\)
    similarly gives \(b=0_A\).
\end{snippetproof}

\begin{snippetproposition}{zero-divisors-simplification}{Cancellation in rings}
    Let \(A\) be a \ring and let \(a\in A\setminus\{0_A\}\).
    \begin{enumerate}
        \item If \(a\) is not a left zero divisor, then
        \[
            ab=ac\implies b=c.
        \]
        \item If \(a\) is not a right zero divisor, then
        \[
            ba=ca\implies b=c.
        \]
    \end{enumerate}
\end{snippetproposition}

\begin{snippetproof}{zero-divisors-simplification-proof}{zero-divisors-simplification}{Cancellation in rings}
    If \(ab=ac\), then \(a(b-c)=0_A\). Since \(a\neq0_A\) is not a left
    zero divisor, \(b-c=0_A\), and hence \(b=c\). The right-hand statement
    is analogous.
\end{snippetproof}

\begin{snippet}{zero-divisor-cancellation-failure}
    Cancellation may fail for a zero divisor: if \(a(b-c)=0_A\) with
    \(a\neq0_A\) and \(b-c\neq0_A\), then \(ab=ac\) although \(b\neq c\).
\end{snippet}

\section{Units}

\begin{snippetdefinition}{ring-unit-group-definition}{Group of units}
    Let \(A\) be a \ring. A \emph{unit} of \(A\) is an element \(a\in A\)
    admitting a two-sided multiplicative inverse. The \set of units is
    denoted
    \[
        A^\times=\{a\in A\suchthat
        \text{there exists }a^{-1}\in A\}.
    \]
\end{snippetdefinition}

\begin{snippetproposition}{ring-units-form-group}{Units form a group}
    Let \(A\) be a \ring. The \set \(A^\times\) is a \group under ring
    multiplication.
\end{snippetproposition}

\begin{snippetproof}{ring-units-form-group-proof}{ring-units-form-group}{Units form a group}
    The identity \(1_A\) is a unit. If \(a,b\in A^\times\), then
    \[
        (ab)^{-1}=b^{-1}a^{-1},
    \]
    so \(ab\in A^\times\). Finally, the inverse of a unit is again a unit.
    Associativity is inherited from \(A\).
\end{snippetproof}

\begin{snippetexample}{ring-unit-groups-example}{Examples of unit groups}
    Let \(K\) be a \field. Then
    \[
        \integers^\times=\{\pm1\},
        \qquad
        K[x]^\times=K^\times,
        \qquad
        \matrices_n(K)^\times=\operatorname{GL}_n(K),
        \qquad
        K^\times=K\setminus\{0_K\}.
    \]
\end{snippetexample}
