\begin{snippetexercise}{rings-ex1}{Boolean rings}
    Let \(A\) be a \ring such that
    \[
        z^2=z
        \qquad
        \text{for every }z\in A.
    \]
    Show that:
    \begin{enumerate}
        \item \(z+z=0_A\) for every \(z\in A\);
        \item \(A\) is commutative.
    \end{enumerate}
\end{snippetexercise}

\begin{snippetsolution}{rings-ex1-sol}{Boolean rings}
    Applying the hypothesis to \(z+z\) gives
    \[
        z+z=(z+z)^2=z^2+z^2+z^2+z^2
        =z+z+z+z.
    \]
    Cancellation in the additive group yields \(z+z=0_A\). Hence
    every element is its own additive inverse.

    For \(a,b\in A\), idempotence of \(a+b\) gives
    \[
        a+b=(a+b)^2=a+ab+ba+b,
    \]
    so \(ab+ba=0_A\), or \(ab=-ba\). Since every element is its own
    additive inverse, \(-ba=ba\), and therefore \(ab=ba\). Thus \(A\)
    is commutative.
\end{snippetsolution}

\begin{snippetnote}{boolean-rings-characteristic-two-note}{Boolean rings have characteristic two}
    In a Boolean \ring,
    \[
        2z=z+z=0_A
        \qquad
        \text{for every }z\in A.
    \]
    This is the step that turns \(ab+ba=0_A\) into \(ab=ba\).
\end{snippetnote}

\begin{snippetexercise}{rings-ex2}{The ring of subsets}
    Let \(U\) be a \set and let \(\mathcal S=\powerset(U)\). For
    \(A,B\subseteq U\), define
    \[
        A+B=(A\difference B)\union(B\difference A),
        \qquad
        A\cdot B=A\intersection B.
    \]
    Show that \((\mathcal S,+,\cdot)\) is a commutative Boolean ring.
\end{snippetexercise}

\begin{snippetsolution}{rings-ex2-sol}{The ring of subsets}
    Associate with every \(A\subseteq U\) its characteristic
    \function
    \[
        \chi_A\colon U\fromto\integers/2\integers.
    \]
    Pointwise,
    \[
        \chi_{A+B}=\chi_A+\chi_B,
        \qquad
        \chi_{A\cdot B}=\chi_A\chi_B.
    \]
    Indeed, the symmetric difference contains precisely the points
    belonging to exactly one of \(A,B\), while intersection corresponds
    to multiplication of the two characteristic values.

    Therefore the operations on \(\powerset(U)\) are transported from
    the pointwise operations on the commutative ring
    \((\integers/2\integers)^U\). The zero and the unit are
    \[
        0_{\mathcal S}=\emptyset,
        \qquad
        1_{\mathcal S}=U.
    \]
    Moreover,
    \[
        A+A=\emptyset,
        \qquad
        A\cdot A=A.
    \]
    Hence every element is idempotent and \(\mathcal S\) is Boolean.
\end{snippetsolution}

\begin{snippetexample}{ring-of-subsets-symmetric-difference-example}{Symmetric difference}
    If \(U=\{1,2,3\}\), then
    \[
        \{1,2\}+\{2,3\}=\{1,3\}.
    \]
    The element \(2\) belongs to both subsets and is therefore removed
    by the symmetric difference.
\end{snippetexample}
