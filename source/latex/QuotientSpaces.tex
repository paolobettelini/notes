\section{Quotient spaces}

\begin{snippet}{vector-space-quotient-motivation}
    Let \(V\) be a \vectorspace over a \field \(\mathbb{F}\), and let
    \(W\subseteq V\) be a linear subspace. The quotient construction
    identifies two \vector[vectors] of \(V\) whenever their difference lies
    in \(W\).
\end{snippet}

\begin{snippetdefinition}{vector-space-subspace-equivalence-relation-definition}{Equivalence modulo a subspace}
    Let \(V\) be a \vectorspace over a \field \(\mathbb{F}\), and let
    \(W\subseteq V\) be a linear subspace. For \(v_1,v_2\in V\), define
    \[
        v_1\sim_W v_2
        \quad \ifandonlyif \quad
        v_1-v_2\in W.
    \]
    In this case, \(v_1\) and \(v_2\) are said to be \emph{equivalent modulo
    \(W\)}.
\end{snippetdefinition}

\begin{snippetproposition}{vector-space-subspace-relation-is-equivalence}{Equivalence modulo a subspace is an equivalence relation}
    Let \(V\) be a \vectorspace over a \field \(\mathbb{F}\), and let
    \(W\subseteq V\) be a linear subspace. The relation \(\sim_W\) defined by
    \[
        v_1\sim_W v_2 \ifandonlyif v_1-v_2\in W
    \]
    is an \equivrelation on \(V\).
\end{snippetproposition}

\begin{snippetproof}{vector-space-subspace-relation-is-equivalence-proof}{vector-space-subspace-relation-is-equivalence}{Equivalence modulo a subspace is an equivalence relation}
    For every \(v\in V\), \(v-v=0_V\in W\), so the relation is reflexive.
    If \(v_1-v_2\in W\), then
    \(v_2-v_1=-(v_1-v_2)\in W\), so it is symmetric. Finally, if
    \(v_1-v_2\in W\) and \(v_2-v_3\in W\), then
    \[
        v_1-v_3=(v_1-v_2)+(v_2-v_3)\in W,
    \]
    so it is transitive.
\end{snippetproof}

\begin{snippet}{vector-space-subspace-equivalence-partition}
    Let \(V\) be a \vectorspace over a \field \(\mathbb{F}\), and let
    \(W\subseteq V\) be a linear subspace. Since \(\sim_W\) is an
    \equivrelation, its equivalence classes form a partition of \(V\).
\end{snippet}

\begin{snippetdefinition}{vector-space-equivalence-class-modulo-subspace-definition}{Equivalence class modulo a subspace}
    Let \(V\) be a \vectorspace over a \field \(\mathbb{F}\), let
    \(W\subseteq V\) be a linear subspace, and let \(v\in V\). The
    \emph{equivalence class of \(v\) modulo \(W\)} is the \set
    \[
        [v]_W
        =\{u\in V\suchthat u\sim_W v\}
        =\{u\in V\suchthat u-v\in W\}.
    \]
\end{snippetdefinition}

\begin{snippetproposition}{vector-space-equivalence-class-is-coset}{Equivalence classes are cosets}
    Let \(V\) be a \vectorspace over a \field \(\mathbb{F}\), let
    \(W\subseteq V\) be a linear subspace, and let \(v\in V\). If
    \([v]_W=\{u\in V\suchthat u-v\in W\}\), then
    \[
        [v]_W=v+W=\{v+w\suchthat w\in W\}.
    \]
\end{snippetproposition}

\begin{snippetproof}{vector-space-equivalence-class-is-coset-proof}{vector-space-equivalence-class-is-coset}{Equivalence classes are cosets}
    For \(u\in V\),
    \[
        u\in[v]_W
        \ifandonlyif u-v\in W
        \ifandonlyif u=v+w\text{ for some }w\in W
        \ifandonlyif u\in v+W.
    \]
\end{snippetproof}

\begin{snippetdefinition}{vector-space-quotient-definition}{Quotient space}
    Let \(V\) be a \vectorspace over a \field \(\mathbb{F}\), and let
    \(W\subseteq V\) be a linear subspace. The \emph{quotient space of \(V\)
    by \(W\)} is the \set
    \[
        V/W=\{[v]_W\suchthat v\in V\}
        =\{v+W\suchthat v\in V\},
    \]
    where
    \[
        [v]_W=v+W=\{v+w\suchthat w\in W\},
    \]
    equipped with the operations
    \[
        (v_1+W)+(v_2+W)=(v_1+v_2)+W,
        \qquad
        \lambda(v+W)=\lambda v+W
    \]
    for \(v,v_1,v_2\in V\) and \(\lambda\in\mathbb{F}\).
\end{snippetdefinition}

\begin{plainhtml}
    Thus the entire subspace W is collapsed to the zero vector of the quotient.
\end{plainhtml}

\begin{snippet}{vector-space-quotient-zero-coset}
    Let \(V\) be a \vectorspace over a \field \(\mathbb{F}\), and let
    \(W\subseteq V\) be a linear subspace. For every \(v\in V\), one has
    \(v\in v+W\). Moreover,
    \[
        0_V+W=W,
    \]
    so every \(w\in W\) represents the zero \vector \(0_{V/W}=W\).
\end{snippet}

\begin{snippetproposition}{vector-space-coset-equality-characterization}{Equality of cosets}
    Let \(V\) be a \vectorspace over a \field \(\mathbb{F}\), let
    \(W\subseteq V\) be a linear subspace, and let \(v_1,v_2\in V\). Then
    \[
        v_1+W=v_2+W
        \ifandonlyif v_1-v_2\in W
        \ifandonlyif v_1\sim_W v_2.
    \]
\end{snippetproposition}

\begin{snippetproof}{vector-space-coset-equality-characterization-proof}{vector-space-coset-equality-characterization}{Equality of cosets}
    Suppose first that \(v_1+W=v_2+W\). Since \(v_1\in v_1+W\), there
    is some \(w\in W\) such that \(v_1=v_2+w\). Hence
    \(v_1-v_2=w\in W\).

    Conversely, suppose that \(v_1-v_2=w\in W\). If \(u\in v_1+W\), then
    \(u=v_1+w'\) for some \(w'\in W\), and therefore
    \[
        u=v_2+(w+w')\in v_2+W.
    \]
    Thus \(v_1+W\subseteq v_2+W\). Since
    \(v_2-v_1=-w\in W\), the reverse inclusion follows in the same way.
\end{snippetproof}

\begin{snippetproposition}{quotient-space-complementary-basis-cosets}{Cosets from an extended basis}
    Let \(V\) be an \(n\)-dimensional \vectorspace over a \field
    \(\mathbb{F}\), let \(W\subseteq V\) be a linear subspace of dimension
    \(r<n\), and suppose that
    \[
        \mathcal{B}_W=\{w_1,\ldots,w_r\}
    \]
    is extended to a \basis
    \[
        \mathcal{B}=\{w_1,\ldots,w_r,v_1,\ldots,v_{n-r}\}
    \]
    of \(V\). Then \(v_i+W\neq v_j+W\) whenever \(i\neq j\). Moreover,
    \(w+W=W\) for every \(w\in W\).
\end{snippetproposition}

\begin{snippetproof}{quotient-space-complementary-basis-cosets-proof}{quotient-space-complementary-basis-cosets}{Cosets from an extended basis}
    If \(v_i+W=v_j+W\) for \(i\neq j\), then \(v_i-v_j\in W\). Hence
    \[
        v_i-v_j=a_1w_1+\cdots+a_rw_r
    \]
    for suitable \(a_1,\ldots,a_r\in\mathbb{F}\), contradicting the
    \linearlyindependent[linear independence] of \(\mathcal{B}\). Finally,
    \(w+W=0_V+W=W\) because \(w-0_V\in W\).
\end{snippetproof}

\begin{plainhtml}
    This construction is the additive analogue of forming a quotient group by a normal subgroup.
\end{plainhtml}
