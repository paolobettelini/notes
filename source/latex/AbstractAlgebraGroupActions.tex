\section{Group actions}

\begin{snippetdefinition}{right-group-action-definition}{(Right) group action}
    Let \(\Omega\) be a non-empty \set and let \((G, \circ)\) be a \group.
    A \emph{right action} of \(G\) on \(\Omega\) is a \function
    \[
        \Omega \times G \fromto \Omega
    \]
    such that:
    \begin{enumerate}
        \item \(\forall x\in \Omega, x1_G = x\);
        \item \(\forall x\in \Omega, \forall g,h \in G, (xg)h = x(gh)\).
    \end{enumerate}
    \emph{Syntax:} \(xg\) denotes the image of the pair \((x, g)\).
\end{snippetdefinition}

\begin{snippetdefinition}{left-group-action-definition}{(Left) group action}
    Let \(\Omega\) be a non-empty \set and let \((G, \circ)\) be a \group.
    A \emph{left action} of \(G\) on \(\Omega\) is a \function
    \[
        G \times \Omega \fromto \Omega
    \]
    such that:
    \begin{enumerate}
        \item \(\forall x\in \Omega, 1_Gx = x\);
        \item \(\forall x\in \Omega, \forall g,h \in G, g(hx) = (gh)x\).
    \end{enumerate}
    \emph{Syntax:} \(gx\) denotes the image of the pair \((g, x)\).
\end{snippetdefinition}

\begin{plainhtml}
    Translations, isometries, rotations with fixed center are all examples of group actions on a plane.
\end{plainhtml}
\begin{plainhtml}
    A group action represents every group element as a permutation of the set on which it acts.
\end{plainhtml}

\begin{snippet}{group-action-restrictions}
    \textbf{Subgroup restriction:} let \(G\) be a \group with an action on a \set \(\Omega\)
    and let \(H \subgroupleq G\). Then,
    an action of \(H\) on \(\Omega\) is naturally induced. \\
    \textbf{Subset restriction:} let \(G\) be a \group with an action on a \set \(\Omega\)
    and let \(\Theta \subseteq \Omega\). Then,
    it is not guaranteed that an action of \(G\) on \(\Theta\) can be induced.
    If the action is closed, then an action is induced.
\end{snippet}

\begin{snippetproposition}{group-action-properties-1}{Group action properties}
    Let \(G\) be a \group with an action on a \set \(\Omega\)
    and let \(g\in G\). Consider the \function \(\varphi_g\)
    from \(\Omega\) to itself which sends \(x\) to \(xg\).
    Then, \(\varphi_g\) is \bijective and
    \({(\varphi_g)}^{-1} = \varphi_{g^{-1}}\).
\end{snippetproposition}

\begin{snippetproof}{group-action-properties-1-proof}{group-action-properties-1}{Group action properties}
    We want to show that \(\varphi_g\) and \(\varphi_{g^{-1}}\) are inverses:
    \begin{align*}
        x(\varphi_g \varphi_{g^{-1}}) &= (x\varphi_g)\varphi_{g^{-1}} = (xg)g^{-1} \\
        &= x(gg^{-1}) = x1_G = x = x\identityfunc
    \end{align*}
    The other direction is analogous.
\end{snippetproof}

\section{Examples}

\begin{snippetdefinition}{regular-right-group-action}{Regular right action}
    Let \((G, \circ)\) be a \group.
    The \function
    \[
        (x, g) \fromto x \circ g
    \]
    is a right group action on the \group itself called \emph{regular right action}.
\end{snippetdefinition}

\begin{plainhtml}
    This action is free and transitive: every element is carried to every other element in exactly one way.
\end{plainhtml}

\begin{snippetdefinition}{conjugate-action-definition}{Conjugate action}
    Let \(G\) be a \group.
    The \emph{conjugation action} is the right action of \(G\) on itself defined by
    \[
        x^g \triangleq g^{-1}xg, \qquad x,g\in G.
    \]
\end{snippetdefinition}

\begin{snippetproposition}{conjugation-is-group-action}{Conjugation is a group action}
    Let \(G\) be a \group. The assignment \(x^g=g^{-1}xg\) is a right action
    of \(G\) on itself, and for every \(g\in G\) the map \(x\mapsto x^g\) is
    an automorphism of \(G\).
\end{snippetproposition}

\begin{snippetproof}{conjugation-is-group-action-proof}{conjugation-is-group-action}{Conjugation is a group action}
    For \(x,g,h\in G\),
    \[
        x^{1_G}=x,
        \qquad (x^g)^h=h^{-1}(g^{-1}xg)h=(gh)^{-1}x(gh)=x^{gh}.
    \]
    Moreover, \((xy)^g=x^gy^g\), and conjugation by \(g^{-1}\) is the
    inverse of conjugation by \(g\). Thus each such map is an automorphism.
\end{snippetproof}

\begin{snippetproposition}{conjugate-group-properties}{Properties of conjugate group}
    Let \(G\) be a \group and let \(x,g,h\in G\).
    \begin{enumerate}
        \item \(x^{1_G} = x\);
        \item \({(x^g)}^h = x^{gh}=(h^{-1}g^{-1})x(gh)\).
    \end{enumerate}
\end{snippetproposition}

\section{Action orbits}

\begin{snippetproposition}{bin-relation-group-action-is-equiv}{}
    Let \(G\) be a \group with an action on a \set \(\Omega\).
    Let \(\sim_G\) be a \binrelation in \(\Omega\) such that
    \(x \sim_G y\) \ifandonlyif there exist \(g\in G\)
    such that \(x=yg\). Then, \(\sim_G\) is an \equivrelation.
\end{snippetproposition}

\begin{snippetproof}{bin-relation-group-action-is-equiv-proof}{bin-relation-group-action-is-equiv}{}
    The \binrelation is an \equivrelation:
    \begin{enumerate}
        \item \emph{reflexive:} \(x \sim_G x\), indeed \(x1_G = x\);
        \item \emph{symmetric:} if \(x \sim_G y\), meaning there exist \(g\in G\)
        such that \(x=yg\), then \(xg^{-1} = (yg)g^{-1} = y(gg^{-1}) = y\).
        Thus, \(y \sim_G x\).
        \item \emph{transitivity:} if \(x \sim_G y\) and \(y\sim_G z\),
        there exist \(h,g \in G\) such that \(x=yg\) and \(y=zh\).
        Thus,
        \[
            x = yg = (zh)g = z(hg)
        \]
        and thus \(x \sim_G z\).
    \end{enumerate}
\end{snippetproof}

\begin{snippetdefinition}{group-action-orbit-definition}{Orbit}
    Let \(G\) be a \group with an action on a \set \(\Omega\).
    The \equivclass[equivalence classes] of \(\sim_G\) defined
    as \(y \sim_G x\) \ifandonlyif there exist \(g\in G\)
    such that \(x=yg\), are called \emph{orbits}. \\
    \emph{Syntax:} let \(y\in \Omega\). The \equivclass of \(y\)
    is formed by the elements of the form \(yg\) for \(g\in G\)
    and denoted \(yG\).
\end{snippetdefinition}

\begin{snippet}{group-action-orbit-expl}
    Given a fixed \(g\in G\), the elements of form \(yg\)
    are all the elements of \(\Omega\) as the function \(y \to yg\)
    is \bijective. \\
    Given a fixed \(y\in \Omega\), the elements of form \(yg\)
    are the ones of an orbit (could be all the elements of \(\Omega\) if there is only an orbit).
\end{snippet}

\subsection{Orbits of actions on groupselves}

\begin{snippetproposition}{regular-right-action-orbit}{Orbits of regualr right action}
    Let \(G\) be a \group and consider the regular action on itself.
    Given \(x\in G\), its orbit is formed by the elements of \(xg\) for \(g\in G\).
    By the cancellation laws, there exist only one orbit.
\end{snippetproposition}

\begin{plainhtml}
    Thus, every element cn be send into any other element.
\end{plainhtml}

\begin{snippetproposition}{conjugate-action-orbit}{Orbits of conjugate action}
    Let \(G\) be a \group and consider the conjugate action on itself.
    There is always an orbit containing only \(1_G\).
    If the \group is not trivial, there are always more orbits,
    which are called \emph{conjugate classes} \(x^G\).
\end{snippetproposition}

\begin{snippetproposition}{group-center-conjugate-class}{Group center conjugate class}
    Let \(G\) be a \group.
    \(x \in \groupcenter(G)\) \ifandonlyif its conjugate class
    contains only \(x\).
\end{snippetproposition}

\section{Stabilizer}

\begin{snippetdefinition}{stabilizer-definition}{Stabilizer}
    Let \(G\) be a \group with an action on \(\Omega\)
    and let \(x\in\Omega\).
    The \emph{stabilizer} of \(x\) is the subset
    \[
        \text{Stab}_G(x) \triangleq \{
            g\in G \suchthat xg = x    
        \}
    \]
\end{snippetdefinition}

\begin{snippetproposition}{stabilizer-is-subgroup}{}
    Let \(G\) be a \group acting on a \set \(\Omega\), and let \(x\in\Omega\). Then,
    \[
        \stabilizer_G(x) \subgroupleq G
    \]
\end{snippetproposition}

\begin{snippetproof}{stabilizer-is-subgroup-proof}{stabilizer-is-subgroup}{}
    \begin{enumerate}
        \item \(1_G \in \stabilizer_G(x)\) as \(x1_G = x\);
        \item if \(g \in \stabilizer_G(x)\) and \(h \in \stabilizer_G(x)\),
        meaning \(xg=x\) and \(xh=x\), then \(x(gh) = (xg)h = xh = x\),
        meaning \(gh \in \stabilizer_G(x)\);
        \item if \(g\in\stabilizer_G(x)\), meaning \(xg = x\), then
        \(xgg^{-1} = xg^{-1}\) and \(xg^{-1} = x\) meaning \(g^{-1} \in \stabilizer_G(x)\).
    \end{enumerate}
\end{snippetproof}

\begin{snippetproposition}{stabilizer-right-coset-equivalence}{}
    Let \(G\) be a \group with an action on \(\Omega\)
    and let \(x\in\Omega\). The following are equivalent
    \begin{enumerate}
        \item \(xg = xh\);
        \item \(\stabilizer_G(x)g = \stabilizer_G(x)h\) meaning \(h\) and \(g\)
        are in the same right coset of the stabilizer.
    \end{enumerate}
\end{snippetproposition}

\begin{snippetproof}{stabilizer-right-coset-equivalence-proof}{stabilizer-right-coset-equivalence}{}
    \begin{iffproof}
        \item From \(xg=xh\) we get \(xgh^{-1} = x\), meaning \(gh^{-1} \in \stabilizer_G(x)\)
        and thus \[
            g = {(gh^{-1})}h \in \stabilizer_G(x)h
            \implies \stabilizer_G(x)g = \stabilizer_G(x)h
        \]
        \item Let \(g = sh\) with \(s\in\stabilizer_G(x)\).
        Then, \(xg = x(sh) = (xs)h = xh\).
    \end{iffproof}
\end{snippetproof}

\begin{snippetcorollary}{stabilizer-index-size}{}
    Let \(G\) be a \group with an action on \(\Omega\)
    and let \(x\in\Omega\). Then,
    \[
        |G\,:\,\stabilizer_G(x)| = |xG|
    \]
\end{snippetcorollary}

\begin{plainhtml}
    The bigger the index, the bigger the orbit.
\end{plainhtml}

\begin{snippetproposition}{stabilizer-cardinality-finite-group}{}
    Let \(G\) be a \group with an action on \(\Omega\)
    where \(G\) is finite. Then,
    \(|xG|\) divides \(|G|\).
\end{snippetproposition}

\begin{snippet}{stabilizer-of-conjugate-action}
    Consider the conjugate action on a \group \(G\).
    We have that
    \[
        \stabilizer_G(x) = \{g \in G \suchthat g^{-1} x g = x\}
    \]
    Note that \(g^{-1} x g = x\) \ifandonlyif \(xg = gx\).
    Thus,
    \[
        \stabilizer_G(x) = \groupcentralizer_G(x)
    \]
    and
    \[
        |x^G| = |G \,:\, \groupcentralizer_G(x)|
    \]
    In particular, \(|x^G| = 1\) \ifandonlyif \(\groupcentralizer_G(x) = G\),
    meaning \(x \in \groupcenter(G)\).
\end{snippet}

\begin{plainhtml}
    Elements with larger centralizers have smaller conjugacy classes. In an abelian group every conjugacy class is a singleton.
\end{plainhtml}

\begin{snippettheorem}{conjugate-classes-equations-theorem}{}
    Let \(G\) be a finite \group. We have
    \[
        |G| = |\groupcenter(G)| + \sum_i c_i
    \]
    where the \(c_i>1\) are the cardinalities of the conjugacy classes
    of non-central elements; each \(c_i\) divides \(|G|\).
\end{snippettheorem}

\begin{snippetproof}{conjugate-classes-equations-theorem-proof}{conjugate-classes-equations-theorem}{}
    Note that \(|G|\) is the sum of the orders
    of the conjugate classes, as they form a partition since they are \equivclass[equivalent classes].
    These cardinalities divide \(|G|\).
    The classes with only one element are the ones in the center,
    and there is one such singleton class for each element of \(\groupcenter(G)\).
\end{snippetproof}

\begin{snippetdefinition}{p-group-definition}{Finite p-group}
    Let \(p\) be prime. A finite \group \(G\) is a \emph{p-group} if its
    order is a power of \(p\).
\end{snippetdefinition}

\begin{snippettheorem}{center-of-p-group-not-trivial-theorem}{Center of p-group is not trivial}
    Let \(G\) be a non-trivial p-group. Then,
    \(\groupcenter(G)\) is not trivial.
\end{snippettheorem}

\begin{snippetproof}{center-of-p-group-not-trivial-theorem-proof}{center-of-p-group-not-trivial-theorem}{Center of p-group is not trivial}
    We have the equation
    \[
        |G| = |\groupcenter(G)| + \sum_i c_i
    \]
    Now, \(|G| = p^n\) for some \(n \geq 1\).
    In particlar, \(|G|\) is a multiple of \(p\).
    Furthermore, each \(c_i\) is a divisor of \(p^n\) greater than \(1\)
    and thus each \(c_i\) is a multiple of \(p\).
    Thus,
    \[
        |\groupcenter(G)| = |G| - \sum_i c_i
    \]
    is a multiple of \(p\) and \(|\groupcenter(G)| \neq 0\) implies that \(|\groupcenter(G)| \geq p\).
    In particular, \(|\groupcenter(G)| \neq 1\). 
\end{snippetproof}

\section{Conjugate subgroup}

\begin{snippetdefinition}{conjugate-subgroup-definition}{The conjugate subgroup}
    Let \(H \subgroupleq G\) be \group[groups]. Then, the \textit{conjugate subgroup} is defined as
    \[
        H^g \triangleq \{
            g^{-1}hg \suchthat h \in H
        \}
    \]
\end{snippetdefinition}

\begin{snippettheorem}{conjugate-subgroup-is-subgroup}{}
    Let \(H \subgroupleq G\), then \(g^{-1}Hg \subgroupleq G\).
\end{snippettheorem}

\begin{snippetproof}{conjugate-subgroup-is-subgroup-proof}{conjugate-subgroup-is-subgroup}{}
    Suppose \(a,b \in g^{-1}Hg\).
    We want to show \(ab^{-1} \in g^{-1}Hg\).\\
    Note that \(a = g^{-1}h_1g\) and \(b = g^{-1}h_2g\)
    for some \(h_1, h_2 \in H\). \\
    This means that \(ab^{-1}=a{(g^{-1}h_2g)}^{-1} = a(g^{-1}h_2^{-1}g)
        =g^{-1}h_1gg^{-1}h_2^{-1}g = g^{-1} (h_1h_2^{-1}) g \in g^{-1}Hg \)
    because \(h_1h_2^{-1} \in H\).
\end{snippetproof}

\begin{snippetdefinition}{normal-subgroup-definition}{Normal subgroup}
    Let \(G\) be a \group and \(H \subgroupleq G\). Then, \(H\)
    is said to be \emph{normal}
    \[
        H \unlhd G
    \]
    if \(h^g \in H\) for all \(h\in H\) and \(g\in G\).
\end{snippetdefinition}

\begin{snippetproposition}{subgroups-of-center-are-normal}{}
    If \(H \subgroupleq \groupcenter(G)\), then \(H \normalsubgrp G\).
    In particular, \(\groupcenter(G) \normalsubgrp G\).
\end{snippetproposition}

\begin{snippetcorollary}{abelian-subgroups-are-normal}{}
    If \(G\) is an \abeliangroup, every \subgroup of \(G\)
    is \normalsubgrptext in \(G\)
\end{snippetcorollary}

\begin{plainhtml}
    We are now going to characterize normal groups
\end{plainhtml}

\begin{snippettheorem}{normal-subgroup-equivalences-theorem}{}
    Let \(H \subgroupleq G\) be \group[groups].
    The following are equivalent:
    \begin{enumerate}
        \item \(H \normalsubgrp G\);
        \item \(H^g \subgroupleq H\) for all \(g\in G\);
        \item \(H^g = H\) for all \(g\in G\);
        \item \(Hg \subseteq gH\) for all \(g\in G\);
        \item \(Hg = gH\) for all \(g\in G\).
    \end{enumerate}
\end{snippettheorem}

\begin{snippetproof}{normal-subgroup-equivalences-theorem-proof}{normal-subgroup-equivalences-theorem}{}
    \begin{itemize}
        \item \((1) \implies (2)\): trivial;
        \item \((3) \implies (2)\): trivial;
        \item \((2) \implies (3)\): we need to show that \(H \subgroupleq H^g\)
        per every \(g\in G\). We know that \(H^g \subgroupleq H\) per every \(g \in G\).
        In particular, \(H^{g{-1}} \subgroupleq H\) for every \(g \in G\).
        But then
        \[
            {\left(H^{g^{-1}}\right)}^g \subgroupleq H^g
        \]
        for every \(g\in G\). However, clearly \({\left(H^{g^{-1}}\right)}^g = H\),
        from which \(H \subgroupleq H^g\) for every \(g\in G\);
        \item \((2) \implies (4)\): we know that \(H^g \subgroupleq H\) for every \(g\in G\),
        meaning \(g^{-1}Hg \subgroupleq H\), from which \(gg^{-1} H g \subseteq gH\),
        meaning \(Hg \subseteq gH\) for every \(g\in G\).
        \item \((4) \implies (5)\): we know that \(Hg \subseteq gH\) for every \(g\in G\).
        We want to show that \(gH \subseteq Hg\) for every \(g\in G\).
        We need to show that given \(g\in G\) and \(h\in H\),
        we have \(gh \in Hg\), meaning \(gh = h'g\) for some \(h' \in H\).
        Now \begin{align*}
            gh = {({(gh)}^{-1})}^{-1} = {(h^{-1} g^{-1})}^{-1}
        \end{align*}
        However, \(h^{-1} g^{-1}\) and \(Hg^{-1} \subseteq g^{-1} H\),
        meaning \(h^{-1}g^{-1} = g^{-1} h''\) for some \(h'' \in H\).
        Thus, \[gh = {(h^{-1}g^{-1})}^{-1} = {(g^{-1}h'')}^{-1} = {(h'')}^{-1} g \in Hg\]
        \item \((4) \implies (2)\): if \(Hg \subseteq gH\) we have \(g^{-1}hg \in g^{-1} g H = H\),
        meaning \(H^g \subgroupleq H\) for very \(g \in G\).
    \end{itemize}
\end{snippetproof}

    \]
\end{snippetcorollary}

\begin{plainhtml}

\section{Normalizer}

\begin{snippetproposition}{conjugation-action-on-subgroups}{Conjugation action on subgroups}
    Let \(G\) be a \group and let \(\operatorname{Sub}(G)\) be the \set of
    all \subgroup[subgroups] of \(G\). The assignment
    \[
        H^g=g^{-1}Hg, \qquad H\in\operatorname{Sub}(G),\ g\in G,
    \]
    defines a right action of \(G\) on \(\operatorname{Sub}(G)\).
\end{snippetproposition}

\begin{snippetproof}{conjugation-action-on-subgroups-proof}{conjugation-action-on-subgroups}{Conjugation action on subgroups}
    A conjugate of a subgroup is a subgroup. Moreover,
    \(H^{1_G}=H\) and \({(H^g)}^h=H^{gh}\), as follows by conjugating
    every element of \(H\).
\end{snippetproof}

\begin{snippetdefinition}{normalizer-group-definition}{Normalizer}
    Let \(G\) be a \group and \(H \subgroupleq G\). Then, the \emph{normalizer}
    is defined as
    \[
        \mathrm{N}_G(H) \triangleq
        \{g\in G \suchthat g^{-1}Hg = H\}.
    \]
\end{snippetdefinition}

\begin{snippettheorem}{normalizer-properties-theorem}{Normalizer properties}
    Let \(G\) be a \group and \(H\subgroupleq G\). Then:
    \begin{enumerate}
        \item \(\groupnormalizer_G(H)\subgroupleq G\);
        \item \(H\normalsubgrp\groupnormalizer_G(H)\);
        \item if \(H\normalsubgrp K\subgroupleq G\), then
            \(K\subgroupleq\groupnormalizer_G(H)\);
        \item \(H\normalsubgrp G\) \ifandonlyif \(\groupnormalizer_G(H)=G\).
    \end{enumerate}
\end{snippettheorem}

\begin{snippetproof}{normalizer-properties-theorem-proof}{normalizer-properties-theorem}{Normalizer properties}
    The normalizer is the stabilizer of \(H\) for the conjugation action on
    subgroups, so it is a subgroup. Its definition says directly that
    \(H\) is normal in it. If \(H\normalsubgrp K\), every \(k\in K\)
    satisfies \(H^k=H\), so \(K\subgroupleq\groupnormalizer_G(H)\).
    Taking \(K=G\) proves the final equivalence.
\end{snippetproof}

\begin{snippettheorem}{number-of-conjugate-subgroups-theorem}{Number of conjugate subgroups}
    Let \(G\) be a \group and \(H\subgroupleq G\). The number of conjugates
    of \(H\) is
    \[
        |G:\groupnormalizer_G(H)|.
    \]
    If \(|G:H|\) is finite, this number is finite and divides \(|G:H|\).
\end{snippettheorem}

\begin{snippetproof}{number-of-conjugate-subgroups-theorem-proof}{number-of-conjugate-subgroups-theorem}{Number of conjugate subgroups}
    The orbit of \(H\) under conjugation is the set of its conjugates, and
    its stabilizer is \(\groupnormalizer_G(H)\). The orbit--stabilizer theorem
    gives the formula. Since \(H\subgroupleq\groupnormalizer_G(H)\), index
    multiplication gives
    \[
        |G:H|=|G:\groupnormalizer_G(H)|
        |\groupnormalizer_G(H):H|,
    \]
    proving the divisibility assertion.
\end{snippetproof}

\section{Normal subgroups and homomorphisms}

\begin{snippetproposition}{group-homomorphism-preserves-conjugation}{Homomorphisms preserve conjugation}
    Let \(\varphi\colon G\fromto K\) be a \grouphomomorphism. For every
    \(x,g\in G\),
    \[
        \varphi(x^g)=\varphi(x)^{\varphi(g)}.
    \]
\end{snippetproposition}

\begin{snippetproof}{group-homomorphism-preserves-conjugation-proof}{group-homomorphism-preserves-conjugation}{Homomorphisms preserve conjugation}
    Directly,
    \[
        \varphi(g^{-1}xg)=\varphi(g)^{-1}\varphi(x)\varphi(g).
    \]
\end{snippetproof}

\begin{snippettheorem}{normal-subgroups-image-preimage-theorem}{Images and preimages of normal subgroups}
    Let \(\varphi\colon G\fromto K\) be a \grouphomomorphism.
    \begin{enumerate}
        \item If \(N\normalsubgrp G\), then \(\varphi(N)\normalsubgrp\image\varphi\).
        \item If \(L\normalsubgrp K\), then
            \(\varphi^{\inversefunction}(L)\normalsubgrp G\).
    \end{enumerate}
\end{snippettheorem}

\begin{snippetproof}{normal-subgroups-image-preimage-theorem-proof}{normal-subgroups-image-preimage-theorem}{Images and preimages of normal subgroups}
    If \(\varphi(n)\in\varphi(N)\) and \(\varphi(g)\in\image\varphi\), then
    \(\varphi(n)^{\varphi(g)}=\varphi(n^g)\in\varphi(N)\). Thus
    \(\varphi(N)\) is normal in the image. If
    \(x\in\varphi^{\inversefunction}(L)\) and \(g\in G\), then
    \(\varphi(x^g)=\varphi(x)^{\varphi(g)}\in L\), so
    \(x^g\in\varphi^{\inversefunction}(L)\).
\end{snippetproof}

\begin{snippetproposition}{product-of-normal-subgroups-is-normal}{Products of normal subgroups}
    Let \(G\) be a \group and let \(N_1,\ldots,N_r\normalsubgrp G\). Then
    \(N_1N_2\cdots N_r\normalsubgrp G\).
\end{snippetproposition}

\begin{snippetproof}{product-of-normal-subgroups-is-normal-proof}{product-of-normal-subgroups-is-normal}{Products of normal subgroups}
    The product of two normal subgroups is a subgroup. Moreover, for every
    \(g\in G\),
    \[
        {(N_1N_2)}^g=N_1^gN_2^g=N_1N_2,
    \]
    so it is normal. Induction gives the finite-product statement.
\end{snippetproof}

\begin{snippetproposition}{normal-subgroup-conjugation-automorphisms}{Conjugation on a normal subgroup}
    Let \(N\normalsubgrp G\). Conjugation restricts to automorphisms of \(N\)
    and determines the right action
    \[
        N\times G\fromto N,\qquad (n,g)\mapsto n^g=g^{-1}ng.
    \]
    Equivalently, it determines a homomorphism from \(G\) to
    \(\operatorname{Aut}(N)\), with the right-action convention for
    composition.
\end{snippetproposition}

\begin{snippetproof}{normal-subgroup-conjugation-automorphisms-proof}{normal-subgroup-conjugation-automorphisms}{Conjugation on a normal subgroup}
    Normality gives \(n^g\in N\). Conjugation by \(g\) is an automorphism,
    with inverse conjugation by \(g^{-1}\), and
    \({(n^g)}^h=n^{gh}\). This is exactly the right-action law and hence the
    corresponding homomorphism law under the stated convention.
\end{snippetproof}
