\section{Sylow theorems}

\begin{snippetdefinition}{sylow-subgroup-definition}{Sylow subgroup}
    Let \(G\) be a finite \group of order \(p^\alpha r\) where \(p\) is \primen and
    \(r \in \integers\) where \(r\) and \(p\) are \coprime.
    Then, a \subgroup of \(G\) of order \(p^\alpha\) is a \emph{Sylow \(p\)-subgroup} of \(G\).
\end{snippetdefinition}

\begin{snippettheorem}{sylow-theorem}{Sylow theorem}
    Let \(G\) be a finite \group of order \(p^\alpha r\) where \(p\) is \primen and
    \(r \in \integers\) where \(r\) and \(p\) are \coprime.
    Let \(n_p\) be the number of \sylowpsubgroup[Sylow \(p\)-subgroups] of \(G\). Then,
    \begin{enumerate}
        \item there exist \sylowpsubgroup[Sylow \(p\)-subgroups] in \(G\);
        \item every \(p\)-subgroup of \(G\) is contained in at least a \sylowpsubgroup;
        \item the \sylowpsubgroup[Sylow \(p\)-subgroups] are conjugated between eachother;
        \item \(n_p\) is a divisor of \(r\) and \(n_p \equiv 1 \pmod{p}\).
    \end{enumerate}
\end{snippettheorem}

\begin{snippetproof}{sylow-theorem-proof}{sylow-theorem}{Sylow theorem}
    Let \(X\) be the \set of subsets of \(G\) of cardinality \(p^\alpha\),
    with the right-translation action. By the fixed-cardinality subsets
    lemma, every orbit containing a subgroup has exactly \(r\) elements and
    contains exactly one subgroup. Hence there are \(n_p\) such orbits.
    Every other orbit has more than \(r\) elements, and its size divides
    \(|G|=p^\alpha r\). A divisor of \(p^\alpha r\) that is not divisible by
    \(p\) divides \(r\), so every orbit larger than \(r\) has size divisible
    by \(p\). Therefore, for some integer \(\xi\),
    \[
        |X|=rn_p+p\xi,
        \qquad |X|\equiv rn_p\pmod p.
    \]

    The cardinality \(|X|=\binom{p^\alpha r}{p^\alpha}\) depends only on
    \(|G|\). Apply the same argument to a cyclic group \(G'\) of order
    \(p^\alpha r\). It has exactly one subgroup of order \(p^\alpha\), so
    \[
        |X|=|X'|\equiv r\pmod p.
    \]
    Since \(r\) is invertible modulo \(p\), it follows that
    \(n_p\equiv1\pmod p\). In particular, \(n_p\neq0\), proving existence.

    Fix a Sylow \(p\)-subgroup \(P\), and let \(Y\) be the set of conjugates
    of \(P\). At this point \(Y\) is only known to be a subset of the set of
    all Sylow \(p\)-subgroups. The normalizer formula gives
    \[
        |Y|=|G:\groupnormalizer_G(P)|\divides|G:P|=r,
    \]
    so \(p\nmid|Y|\).

    Let \(H\) be any \(p\)-subgroup of \(G\), and let \(H\) act on \(Y\)
    by conjugation. Every orbit has cardinality dividing \(|H|\), hence is
    either a singleton or has cardinality divisible by \(p\). Since
    \(p\nmid|Y|\), some \(Q=P^x\in Y\) is fixed by all of \(H\). Equivalently,
    \[
        H\subgroupleq\groupnormalizer_G(Q).
    \]
    Now \(Q\normalsubgrp\groupnormalizer_G(Q)\), so \(QH\) is a subgroup.
    It is a \(p\)-group containing the Sylow subgroup \(Q\), and therefore
    \(QH=Q\). Thus \(H\subgroupleq Q\), proving that every \(p\)-subgroup is
    contained in a Sylow subgroup.

    If \(H\) itself is Sylow, then \(H\subgroupleq Q\) and the two groups
    have the same order, so \(H=Q=P^x\). Hence all Sylow \(p\)-subgroups
    are conjugate, \(Y\) is the full set of them, and
    \(n_p=|Y|\divides r\).
\end{snippetproof}

\begin{snippetcorollary}{cauchy-lemma}{Cauchy's theorem}
    Let \(G\) be a finite \group and let \(p\) be a prime dividing \(|G|\).
    Then \(G\) contains an element, and hence a \subgroup, of order \(p\).
\end{snippetcorollary}

\begin{snippetproof}{cauchy-lemma-proof}{cauchy-lemma}{Cauchy's theorem}
    A Sylow \(p\)-subgroup \(P\) exists. The normal subgroup ladder for
    finite \(p\)-groups gives a subgroup of \(P\), and hence of \(G\), of
    order \(p\). Any non-identity element of that subgroup has order \(p\).
\end{snippetproof}

\begin{snippetcorollary}{finite-group-subgroups-prime-power-orders}{Subgroups of prime-power order}
    Let \(G\) be a finite \group. If \(p^\beta\divides|G|\), where \(p\) is
    prime, then \(G\) has a subgroup of order \(p^\beta\).
\end{snippetcorollary}

\begin{snippetproof}{finite-group-subgroups-prime-power-orders-proof}{finite-group-subgroups-prime-power-orders}{Subgroups of prime-power order}
    Let \(P\) be a Sylow \(p\)-subgroup of \(G\). Its order is \(p^\alpha\)
    for some \(\alpha\geq\beta\). The normal subgroup ladder in \(P\) gives
    a subgroup of order \(p^\beta\).
\end{snippetproof}
