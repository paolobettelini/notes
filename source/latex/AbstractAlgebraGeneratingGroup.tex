\section{Generating subgroup}

\begin{snippet}{group-generator-expl1}
    The union of two \set[sets] is the biggest subset that contains both \set[sets].
    Given a \group \((G, \circ)\) and \(X \subseteq G\), we want to find a
    \subgroup of \(G\) that contains \(X\). Trivially, \((G, \circ)\) does.
    However, we focus on the smallest possible \subgroup containing \(X\)
    (i.e. contained in every \subgroup[subgroups] containing \(X\)).
    We consider the intersection of all \subgroup[subgroups]
    of \((G, \circ)\) containing \(X\) (which is non-empty, as we pointed out).
    This construction is a \subgroup containing \(X\) and it is the smallest doing so.
\end{snippet}

\begin{snippetdefinition}{generating-set-of-group-definition}{Generating set of a group}
    Let \((G, \circ)\) be a \group and \(X\subseteq G\).
    The \textit{subgroup generated by} \(X\) is the smallest \subgroup \(H\) of \((G, \circ)\)
    such that \(X\subseteq H\).
    \[
        \langle X \rangle = \left(\, \bigcap_{\substack{(H, \circ) \subgroupleq (G,\circ) \\ X \subseteq H}} H, \circ\right)
    \]
    \textit{Syntax:} \(\langle \{x_1, x_2, \cdots, x_n\} \rangle \triangleq \langle x_1, x_2, \cdots, x_n \rangle\).
\end{snippetdefinition}

\begin{snippetproposition}{generated-subgroup-minimality-properties}{Generated subgroup minimality properties}
    Let \(G\) be a \group and let \(X,Y\subseteq G\). Then:
    \begin{enumerate}
        \item \(X\subseteq\gengrp{X}\);
        \item if \(X\subseteq H\subgroupleq G\), then \(\gengrp{X}\subgroupleq H\);
        \item if \(X\subseteq Y\), then \(\gengrp{X}\subgroupleq\gengrp{Y}\);
        \item \(\gengrp{X\union Y}=\gengrp{\gengrp{X}\union\gengrp{Y}}\).
    \end{enumerate}
\end{snippetproposition}

\begin{snippetproof}{generated-subgroup-minimality-properties-proof}{generated-subgroup-minimality-properties}{Generated subgroup minimality properties}
    The first two assertions are the defining properties of the intersection
    of all subgroups containing \(X\). If \(X\subseteq Y\), the subgroup
    \(\gengrp{Y}\) contains \(X\), so minimality gives the third assertion.
    Both sides of the final equality are the smallest subgroup containing
    \(X\union Y\).
\end{snippetproof}

\begin{snippet}{group-generator-expl2}
    Note that \(X \subgroupleq G \implies \gengrp{X} = X\).
    We now want to describe the elements of \(\gengrp{X}\).
    If \(X = \{x,y,z\} \neq \emptyset\), then \(\gengrp{X}\) must contain the products and the inverses
    of the elements of \(X\) (as well as the other elements in \(X\)).
    These are \(x\circ y, y\circ z, \cdots, x^{-1}, z^{-1}, \cdots\) but also
    the products and the inverses of the elements found
    \(x\circ y \circ y \circ z, x \circ y \circ z^{-1}, \cdots\).
    All of these elements are sufficient.
\end{snippet}

\begin{snippettheorem}{generated-subgroups-contains-powers-theorem}{}
    Let \((G, \circ)\) be a \group and let \(X \subseteq G\).
    Then, if \(X = \emptyset\), \[
        \gengrp{X} = (\{1\}, \circ)
    \]
    Otherwise, \(\gengrp{X}\)
    is formed by the elements
    \[
        x_1^{\varepsilon_1} \circ x_2^{\varepsilon_2} \circ \cdots \circ x_n^{\varepsilon_n}
    \]
    where \(n\geq1\), \(x_i\in X\), and
    \(\varepsilon_i\in\{-1,1\}\).
\end{snippettheorem}

\begin{snippetproof}{generated-subgroups-contains-powers-proof}{generated-subgroups-contains-powers-theorem}{}
    The case \(X=\emptyset\) is trivial.
    Let \(X \neq \emptyset\) and \(W\) be the \set formed by elements of the form
    \[
        x_1^{\varepsilon_1} \circ x_2^{\varepsilon_2} \circ \cdots \circ x_n^{\varepsilon_n}
    \]
    with \(x_i\in X\) and \(\varepsilon_i\in\{-1,1\}\).
    Clearly, the elements of \(W\) must be in every \subgroup[subgroups] containing \(X\).
    Thus, \(W \subseteq \gengrp{X}\).
    On the other hand, \(X \subseteq W\).
    We need to show that \((W, \circ)\) is a \subgroup.
    If two words belong to \(W\), their concatenation is again a word of the
    same form, so \(W\) is closed under multiplication. Moreover,
    \[
        {(x_1^{\varepsilon_1}\circ\cdots\circ x_n^{\varepsilon_n})}^{-1}
        =x_n^{-\varepsilon_n}\circ\cdots\circ x_1^{-\varepsilon_1}\in W.
    \]
    Thus \(W\) is a subgroup containing \(X\), so
    \(\gengrp{X}\subgroupleq W\). Together with the reverse inclusion this
    proves equality.
\end{snippetproof}
