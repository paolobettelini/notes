\section{Cantor Sets}

\begin{snippetdefinition}{cantor-set-definition}{Cantor set}
    Let \(C_0=[0,1]\). Given \(C_n\), obtain \(C_{n+1}\) by removing the open
    middle third of every interval composing \(C_n\). Thus
    \[
        C_1=\left[0,\frac13\right]\cup\left[\frac23,1\right]
    \]
    and
    \[
        C_2=
        \left[0,\frac19\right]
        \cup\left[\frac29,\frac13\right]
        \cup\left[\frac23,\frac79\right]
        \cup\left[\frac89,1\right].
    \]
    The \emph{ternary Cantor set} is
    \[
        C\triangleq\bigcap_{n=0}^{\infty}C_n.
    \]
\end{snippetdefinition}

\begin{snippetproposition}{cantor-set-ternary-expansion}{Ternary characterization of the Cantor set}
    A point \(x\in[0,1]\) belongs to \(C\) if and only if it has a base-three
    expansion using only the digits \(0\) and \(2\).
\end{snippetproposition}

\begin{snippet}{cantor-set-ternary-endpoints}
    Ternary expansions are not always unique. For example,
    \(1/3=0.1_3=0.0\overline2_3\). Thus endpoints which have a terminating
    expansion containing a digit \(1\) also possess an expansion using only
    \(0\) and \(2\), as required by the characterization.
\end{snippet}

\begin{snippetproposition}{cantor-set-perfect}{The Cantor set is perfect}
    The Cantor set \(C\) is closed and has no isolated points.
\end{snippetproposition}

\begin{snippetproof}{cantor-set-perfect-proof}{cantor-set-perfect}{The Cantor set is perfect}
    The set \(C\) is closed because it is an intersection of closed sets.
    Let \(x\in C\) and \(\varepsilon>0\). Choose \(n\) such that
    \(3^{-n}<\varepsilon\). The point \(x\) lies in one of the closed
    intervals composing \(C_n\), whose length is \(3^{-n}\). At least one
    endpoint \(y\) of that interval differs from \(x\). Every such endpoint
    belongs to \(C\), and
    \[
        0<|x-y|\leq3^{-n}<\varepsilon.
    \]
    Hence every point of \(C\) is an accumulation point.
\end{snippetproof}

\begin{snippettheorem}{nonempty-perfect-real-set-uncountable}{Nonempty perfect subsets of the real line are uncountable}
    Every nonempty perfect subset of \(\realnumbers\) has the cardinality of
    the continuum. In particular, the Cantor set is uncountable.
\end{snippettheorem}

\begin{snippetproposition}{cantor-set-lebesgue-measure-zero}{The Cantor set has measure zero}
    The ternary Cantor set is compact, has empty interior, and
    \[
        \mu(C)=0.
    \]
\end{snippetproposition}

\begin{snippetproof}{cantor-set-lebesgue-measure-zero-proof}{cantor-set-lebesgue-measure-zero}{The Cantor set has measure zero}
    At stage \(n\geq1\), one removes \(2^{n-1}\) intervals of length
    \(3^{-n}\). They are pairwise disjoint, so
    \[
        \mu([0,1]\setminus C)
        =\sum_{n=1}^{\infty}\frac{2^{n-1}}{3^n}
        =\frac12\sum_{n=1}^{\infty}\left(\frac23\right)^n
        =1.
    \]
    Therefore \(\mu(C)=0\). In particular, \(C\) is both Lebesgue and
    Peano--Jordan measurable, despite having continuum cardinality.
\end{snippetproof}

\section{Fat Cantor Sets}

\begin{snippetdefinition}{fat-cantor-set-definition}{Fat Cantor set}
    Fix \(0<\lambda\leq1/3\). Start with
    \(C_{\lambda,0}=[0,1]\). At stage \(n\geq1\), remove from the center of
    each remaining interval an open interval of length \(\lambda^n\), and
    denote the remaining compact set by \(C_{\lambda,n}\). The
    \emph{fat Cantor set} with parameter \(\lambda\) is
    \[
        C_\lambda=\bigcap_{n=0}^{\infty}C_{\lambda,n}.
    \]
    For \(\lambda=1/3\), this is the ternary Cantor set.
\end{snippetdefinition}

\begin{snippetproposition}{fat-cantor-set-measure}{Measure of a fat Cantor set}
    For \(0<\lambda\leq1/3\),
    \[
        \mu(C_\lambda)=\frac{1-3\lambda}{1-2\lambda}.
    \]
    In particular, \(\mu(C_\lambda)>0\) when \(\lambda<1/3\).
\end{snippetproposition}

\begin{snippetproof}{fat-cantor-set-measure-proof}{fat-cantor-set-measure}{Measure of a fat Cantor set}
    At stage \(n\), one removes \(2^{n-1}\) intervals of length
    \(\lambda^n\). Hence
    \[
        \mu([0,1]\setminus C_\lambda)
        =\sum_{n=1}^{\infty}2^{n-1}\lambda^n
        =\frac12\sum_{n=1}^{\infty}(2\lambda)^n
        =\frac{\lambda}{1-2\lambda}.
    \]
    Subtracting this quantity from one gives the formula.
\end{snippetproof}

\begin{snippetproposition}{fat-cantor-set-contains-no-intervals}{A fat Cantor set contains no interval}
    The set \(C_\lambda\) has empty interior and contains no nondegenerate
    interval.
\end{snippetproposition}

\begin{snippetproof}{fat-cantor-set-contains-no-intervals-proof}{fat-cantor-set-contains-no-intervals}{A fat Cantor set contains no interval}
    The lengths of the intervals composing \(C_{\lambda,n}\) tend to zero.
    Given distinct \(x,y\in C_\lambda\), for all sufficiently large \(n\)
    these two points cannot lie in the same component interval of
    \(C_{\lambda,n}\). Between their components lies an interval removed at
    some stage, and hence a point outside \(C_\lambda\). Thus no interval is
    contained in \(C_\lambda\).
\end{snippetproof}

\begin{snippettheorem}{fat-cantor-indicator-no-riemann-representative}{A Lebesgue class with no Riemann-integrable representative}
    Let \(0<\lambda<1/3\). The indicator
    \(\chi_{C_\lambda}\) belongs to \(L^1([0,1])\), but no function equal to
    it almost everywhere is Riemann integrable on \([0,1]\).
\end{snippettheorem}

\begin{snippetproof}{fat-cantor-indicator-no-riemann-representative-proof}{fat-cantor-indicator-no-riemann-representative}{A Lebesgue class with no Riemann-integrable representative}
    The set \(C_\lambda\) is measurable and has positive finite measure, so
    its indicator is integrable. It is continuous on the open complement of
    \(C_\lambda\) and discontinuous at every point of \(C_\lambda\), because
    the set is perfect and contains no interval.

    Let \(h=\chi_{C_\lambda}\) outside a null set \(N\). For every
    \(x\in C_\lambda\setminus N\), every neighborhood of \(x\) contains an
    open interval in the complement of \(C_\lambda\). Such an interval
    cannot be contained in \(N\), so it contains points where \(h=0\), while
    \(h(x)=1\). Thus \(h\) is discontinuous at every point of
    \(C_\lambda\setminus N\), a set of positive measure. Lebesgue's criterion
    therefore shows that \(h\) is not Riemann integrable.
\end{snippetproof}

\begin{snippet}{fat-cantor-set-measurability-consequences}
    A positive-measure fat Cantor set contains nonmeasurable subsets. It is
    compact but not Peano--Jordan measurable: its boundary is the set itself
    and has positive measure.
\end{snippet}
